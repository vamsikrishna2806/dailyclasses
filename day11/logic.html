2. app/books.py: turning a file into a searchable book

book_genai_studio/app/books.py
What it does

Reads PDF, EPUB, DOCX, TXT, MD and HTML files, finds the chapters, and cuts the text into small overlapping chunks.

The logic, step by step
  1. extract(filename, data) picks a reader based on the file extension and returns a list of pages (and the title, if the file has one).
  2. clean(text) fixes spacing and broken lines.
  3. split_chapters(text) uses a regular expression (CHAPTER_RE) to find headings like "Chapter 3" or "CHAPTER TWELVE". If there are none, it makes equal-sized "Parts".
  4. make_chunks(...) cuts the text into pieces of about 450 tokens, with 80 tokens of overlap, so an idea isn't lost at a chunk boundary. Each chunk remembers its chapter and page.
  5. build() checks the file size, empty files, scanned PDFs (no text) and the token limit, each with a friendly error, and returns the book as a dictionary.
  6. Store keeps books in memory, with a JSON copy on disk.

"""Turn an uploaded book into clean text, chapters and searchable chunks.

Pipeline:  file bytes -> pages of text -> cleaned full text -> chapters -> chunks (~450 tokens)
"""
import html
import io
import json
import re
import time
import uuid
import zipfile
from pathlib import Path

from . import config
from .tokens import count_tokens

SUPPORTED = {".txt", ".md", ".pdf", ".docx", ".epub", ".html", ".htm"}
BOOKS_DIR = config.DATA_DIR / "books"
BOOKS_DIR.mkdir(parents=True, exist_ok=True)

CHUNK_TOKENS = 450
OVERLAP_TOKENS = 80


class BookError(Exception):
    def __init__(self, status: int, code: str, message: str, hint: str = ""):
        super().__init__(message)
        self.status, self.code, self.message, self.hint = status, code, message, hint

    def to_dict(self):
        return {"error": {"code": self.code, "message": self.message, "hint": self.hint}}


# ---------------------------------------------------------------- text extraction

def _decode(data: bytes) -> str:
    for enc in ("utf-8-sig", "utf-16", "cp1252", "latin-1"):
        try:
            text = data.decode(enc)
            if enc == "utf-16" and "\x00" in text:
                continue
            return text
        except UnicodeDecodeError:
            continue
    return data.decode("utf-8", errors="replace")


def _html_to_text(markup: str) -> str:
    markup = re.sub(r"(?is)<(script|style|head)\b.*?</\1>", " ", markup)
    markup = re.sub(r"(?is)<h[1-2][^>]*>(.*?)</h[1-2]>", lambda m: "\n\n# " + re.sub(r"<[^>]+>", "", m.group(1)).strip() + "\n\n", markup)
    markup = re.sub(r"(?i)<br\s*/?>", "\n", markup)
    markup = re.sub(r"(?i)</(p|div|li|h[3-6]|tr|blockquote|section)>", "\n\n", markup)
    markup = re.sub(r"<[^>]+>", "", markup)
    return html.unescape(markup)


def _epub_pages(data: bytes) -> tuple[list[str], str | None]:
    try:
        z = zipfile.ZipFile(io.BytesIO(data))
    except zipfile.BadZipFile:
        raise BookError(422, "bad_epub", "This EPUB file is damaged.")
    title = None
    order = []
    try:
        container = z.read("META-INF/container.xml").decode("utf-8", "replace")
        opf_path = re.search(r'full-path="([^"]+)"', container).group(1)
        opf = z.read(opf_path).decode("utf-8", "replace")
        base = opf_path.rsplit("/", 1)[0] + "/" if "/" in opf_path else ""
        t = re.search(r"<dc:title[^>]*>(.*?)</dc:title>", opf, re.S)
        title = html.unescape(t.group(1).strip()) if t else None
        manifest = {m.group(1): m.group(2) for m in re.finditer(r'<item\b[^>]*?id="([^"]+)"[^>]*?href="([^"]+)"', opf)}
        manifest.update({m.group(2): m.group(1) for m in re.finditer(r'<item\b[^>]*?href="([^"]+)"[^>]*?id="([^"]+)"', opf)})
        for idref in re.findall(r'<itemref\b[^>]*idref="([^"]+)"', opf):
            if idref in manifest:
                order.append(base + manifest[idref])
    except (KeyError, AttributeError):
        pass
    if not order:
        order = sorted(n for n in z.namelist() if n.lower().endswith((".xhtml", ".html", ".htm")))
    pages = []
    for name in order:
        try:
            pages.append(_html_to_text(z.read(name).decode("utf-8", "replace")))
        except KeyError:
            continue
    return pages, title


def extract(filename: str, data: bytes) -> tuple[list[str], str | None, bool]:
    """Returns (pages, title_from_metadata, has_real_page_numbers)."""
    ext = Path(filename).suffix.lower()
    if ext not in SUPPORTED:
        raise BookError(415, "unsupported_type", f"“{ext or 'no extension'}” files aren't supported.",
                        "Upload PDF, EPUB, DOCX, TXT, MD or HTML.")
    if ext in (".txt", ".md"):
        return [_decode(data)], None, False
    if ext in (".html", ".htm"):
        return [_html_to_text(_decode(data))], None, False
    if ext == ".epub":
        pages, title = _epub_pages(data)
        return pages, title, False
    if ext == ".docx":
        try:
            from docx import Document
            doc = Document(io.BytesIO(data))
        except Exception:
            raise BookError(422, "bad_docx", "Couldn't open this Word file.")
        lines = []
        for p in doc.paragraphs:
            style = (p.style.name if p.style is not None else "") or ""
            lines.append(("# " + p.text) if style.startswith(("Heading 1", "Title")) and p.text.strip() else p.text)
        title = doc.core_properties.title or None
        return ["\n\n".join(lines)], title, False
    # PDF
    try:
        from pypdf import PdfReader
        reader = PdfReader(io.BytesIO(data))
        if reader.is_encrypted:
            reader.decrypt("")
        pages = [(p.extract_text() or "") for p in reader.pages]
        title = (reader.metadata.title if reader.metadata else None) or None
    except Exception:
        raise BookError(422, "bad_pdf", "Couldn't read this PDF (damaged or password-protected).")
    return pages, title, True


def clean(text: str) -> str:
    text = text.replace("\r\n", "\n").replace("\r", "\n").replace("­", "")
    text = re.sub(r"(\w)-\n(\w)", r"\1\2", text)        # join words split across lines
    text = re.sub(r"[ \t ]+", " ", text)
    text = re.sub(r" *\n *", "\n", text)
    text = re.sub(r"\n{3,}", "\n\n", text)
    return text.strip()


# ---------------------------------------------------------------- chapters

_NUM = r"(?:\d+|[ivxlcdm]+|one|two|three|four|five|six|seven|eight|nine|ten|eleven|twelve|[a-z]+teen|twenty[a-z\- ]*|thirty[a-z\- ]*)"
CHAPTER_RE = re.compile(
    rf"^(?:#{{1,2}} .{{1,100}}|(?:chapter|part|book|section|act|lesson|unit) {_NUM}\b.{{0,80}}"
    r"|prologue|epilogue|introduction|preface|foreword|afterword|conclusion|appendix\b.{0,60})$",
    re.I | re.M,
)


def split_chapters(text: str) -> list[dict]:
    marks = [(m.start(), m.group(0).lstrip("# ").strip()) for m in CHAPTER_RE.finditer(text)]
    sections = []
    if marks and marks[0][0] > 0:
        sections.append(["Opening", 0])
    for start, title in marks:
        sections.append([title[:100], start])
    if not sections:
        sections = [["Opening", 0]]
    bounds = []
    for i, (title, start) in enumerate(sections):
        end = sections[i + 1][1] if i + 1 < len(sections) else len(text)
        bounds.append({"title": title, "start": start, "end": end})

    # Merge tiny sections (table-of-contents lines, stray headings) into the next one.
    merged = []
    carry_start = None
    for b in bounds:
        words = len(text[b["start"]:b["end"]].split())
        start = carry_start if carry_start is not None else b["start"]
        if words < 120 and b is not bounds[-1]:
            carry_start = start
            continue
        merged.append({"title": b["title"], "start": start, "end": b["end"]})
        carry_start = None

    # No real chapters? Make even-sized parts so "summarize by chapter" still works.
    if len(merged) < 2 and count_tokens(text) > 6000:
        merged, size, pos, n = [], 12000 * 4, 0, 1  # ~3000 words per part
        while pos < len(text):
            end = min(len(text), pos + size)
            if end < len(text):
                nl = text.rfind("\n\n", pos + size // 2, end)
                end = nl if nl > 0 else end
            merged.append({"title": f"Part {n}", "start": pos, "end": end})
            pos, n = end, n + 1
    return merged


# ---------------------------------------------------------------- chunks

_SENT_RE = re.compile(r"(?<=[.!?…”\"])\s+(?=[A-Z“\"'(\[])")


def split_sentences(text: str) -> list[str]:
    """Sentences, with short unpunctuated lines (titles, headings) kept as their own items."""
    out = []
    for para in re.split(r"\n\s*\n", text):
        lines, buf = para.split("\n"), []
        for line in lines:
            s = line.strip()
            if s and len(s.split()) <= 12 and not re.search(r"[.!?…:;,”\"')]$", s):
                if buf:
                    out.extend(_SENT_RE.split(" ".join(buf)))
                    buf = []
                out.append(s)
            elif s:
                buf.append(s)
        if buf:
            out.extend(_SENT_RE.split(" ".join(buf)))
    return [s.strip() for s in out if s.strip()]


def _pieces(text: str) -> list[str]:
    """Paragraphs, with very long paragraphs broken into sentence groups."""
    out = []
    for para in re.split(r"\n\s*\n", text):
        para = para.strip()
        if not para:
            continue
        if count_tokens(para) <= CHUNK_TOKENS:
            out.append(para)
            continue
        buf = ""
        for s in split_sentences(para):
            if buf and count_tokens(buf + " " + s) > CHUNK_TOKENS:
                out.append(buf)
                buf = s
            else:
                buf = (buf + " " + s).strip()
        if buf:
            out.append(buf)
    return out


def make_chunks(text: str, chapters: list[dict], page_starts: list[int] | None) -> list[dict]:
    chunks = []
    for ci, ch in enumerate(chapters):
        body = text[ch["start"]:ch["end"]]
        buf, buf_tokens, cursor = [], 0, ch["start"]
        pieces = _pieces(body)

        def flush():
            nonlocal cursor
            chunk_text = "\n\n".join(buf)
            pos = text.find(buf[0][:60], cursor)
            if pos >= 0:
                cursor = pos
            page = None
            if page_starts:
                page = sum(1 for s in page_starts if s <= max(pos, ch["start"]))
            chunks.append({"id": len(chunks), "chapter": ci, "page": page,
                           "tokens": count_tokens(chunk_text), "text": chunk_text})

        for p in pieces:
            t = count_tokens(p)
            if buf and buf_tokens + t > CHUNK_TOKENS:
                flush()
                last = buf[-1]
                lt = count_tokens(last)
                buf, buf_tokens = ([last], lt) if lt <= OVERLAP_TOKENS * 2 else ([], 0)
            buf.append(p)
            buf_tokens += t
        if buf:
            flush()
    return chunks

def build(filename: str, data: bytes) -> dict:
    if len(data) > config.MAX_UPLOAD_MB * 1024 * 1024:
        raise BookError(413, "file_too_large", f"File is over {config.MAX_UPLOAD_MB:g} MB.",
                        "Compress the PDF or upload a .txt version." +
                        (" (Vercel's free plan limits uploads to about 4.5 MB.)" if config.ON_VERCEL else ""))
    if not data:
        raise BookError(400, "empty_file", "The file is empty.")
    pages, meta_title, paged = extract(filename, data)
    cleaned = [clean(p) for p in pages]
    text, page_starts = "", []
    for p in cleaned:
        if not p:
            page_starts.append(len(text))
            continue
        if text:
            text += "\n\n"
        page_starts.append(len(text))
        text += p
    if len(text.split()) < 50:
        raise BookError(422, "no_text", "No readable text was found in this file.",
                        "If it's a scanned PDF, run OCR on it first (e.g. with Google Drive or Adobe).")
    n_tokens = count_tokens(text)
    if n_tokens > config.MAX_BOOK_TOKENS:
        raise BookError(413, "book_too_long", f"This book has about {n_tokens:,} tokens; the limit is "
                        f"{config.MAX_BOOK_TOKENS:,}.", "Split it into volumes or raise MAX_BOOK_TOKENS.")
    chapters = split_chapters(text)
    chunks = make_chunks(text, chapters, page_starts if paged else None)
    for i, ch in enumerate(chapters):
        ch["idx"] = i
        ch["tokens"] = sum(c["tokens"] for c in chunks if c["chapter"] == i)
        ch["words"] = len(text[ch["start"]:ch["end"]].split())
    title = (meta_title or "").strip() or Path(filename).stem.replace("_", " ").replace("-", " ").strip().title()
    return {
        "id": uuid.uuid4().hex[:12],
        "title": title[:120],
        "filename": filename,
        "created": time.time(),
        "pages": len(pages) if paged else None,
        "words": len(text.split()),
        "tokens": n_tokens,
        "text": text,
        "chapters": chapters,
        "chunks": chunks,
        "embeddings": None,
        "cache": {},
    }


def public(book: dict) -> dict:
    """Book info without the big fields."""
    return {k: book[k] for k in ("id", "title", "filename", "created", "pages", "words", "tokens")} | {
        "chapters": [{k: c[k] for k in ("idx", "title", "tokens", "words")} for c in book["chapters"]],
        "chunks": len(book["chunks"]),
        "semantic_search": book.get("embeddings") is not None,
    }


def portable(book: dict) -> dict:
    """Everything needed to rebuild the book on another server (no embeddings: too big)."""
    return {k: book[k] for k in ("id", "title", "filename", "created", "pages", "words", "tokens",
                                 "text", "chapters", "chunks")}


class Store:
    """Books live in memory, with a JSON copy on disk."""

    def __init__(self):
        self.books: dict[str, dict] = {}

    def _path(self, book_id):
        return BOOKS_DIR / f"{re.sub(r'[^a-f0-9]', '', book_id)}.json"

    def save(self, book: dict):
        self.books[book["id"]] = book
        try:
            self._path(book["id"]).write_text(json.dumps(book), encoding="utf-8")
        except OSError:
            pass  # disk is optional (e.g. read-only); memory copy still works

    def get(self, book_id: str) -> dict:
        if book_id in self.books:
            return self.books[book_id]
        p = self._path(book_id)
        if p.exists():
            book = json.loads(p.read_text(encoding="utf-8"))
            self.books[book_id] = book
            return book
        raise BookError(404, "book_not_found", "This server doesn't have that book any more.",
                        "Your browser has a copy; it will be re-sent automatically.")

    def all(self) -> list[dict]:
        for p in BOOKS_DIR.glob("*.json"):
            if p.stem not in self.books:
                try:
                    self.books[p.stem] = json.loads(p.read_text(encoding="utf-8"))
                except (OSError, ValueError):
                    continue
        return sorted(self.books.values(), key=lambda b: b["created"], reverse=True)

    def delete(self, book_id: str):
        self.books.pop(book_id, None)
        try:
            self._path(book_id).unlink(missing_ok=True)
        except OSError:
            pass

3. app/retrieval.py: finding the right passages

book_genai_studio/app/retrieval.py
What it does

Given a question, it finds the 6 chunks of the book most likely to contain the answer.

The logic, step by step
  1. tokenize() lower-cases the text, removes stop words (the, a, is…) and trims word endings, so "climbing" matches "climb".
  2. class BM25 is a classic keyword search: it scores chunks that contain the question's rare words more often.
  3. search() ranks chunks by BM25. If the book has embeddings, it also ranks them by meaning (cosine similarity of vectors).
  4. The two rankings are merged with Reciprocal Rank Fusion: each chunk gets 1/(60+rank) from each list, and the top k win.
  5. extractive_summary, best_sentences and keywords are the offline Demo-mode tools: they pick important sentences without any AI.

"""Find the passages of a book that best match a question.

Two searches, combined:
  * BM25 keyword search  - always available, free, good for names and exact terms
  * Embedding search     - needs an OpenAI key, good for meaning ("why was she angry?")
Results are merged with Reciprocal Rank Fusion (RRF): a passage ranked high by
either search ends up near the top.
"""
import math
import re
from collections import Counter

import numpy as np

WORD = re.compile(r"[a-z0-9]+(?:'[a-z]+)?")
STOP = set("""a about above after again against all am an and any are as at be because been before being
below between both but by can could did do does doing down during each few for from further had has have
having he her here hers herself him himself his how i if in into is it its itself just me more most my myself
no nor not now of off on once only or other our ours ourselves out over own same she should so some such than
that the their theirs them themselves then there these they this those through to too under until up very was
we were what when where which while who whom why will with would you your yours yourself yourselves also
said says say one may might shall upon unto thee thou thy""".split())


INITIALS = re.compile(r"\b(?:[a-z]\.){2,}")


def _stem(w: str) -> str:
    """Tiny stemmer so 'logbooks', 'climbed', 'climbing' match 'logbook', 'climb'."""
    for suf in ("ing", "ed", "es", "s"):
        if w.endswith(suf) and len(w) - len(suf) >= 4 and not w.endswith("ss"):
            return w[: -len(suf)]
    return w


def tokenize(text: str) -> list[str]:
    low = text.lower()
    words = [_stem(w) for w in WORD.findall(low) if w not in STOP and len(w) > 1]
    return words + [i.replace(".", "") + "_init" for i in INITIALS.findall(low)]  # keep "A.O." as one token


class BM25:
    def __init__(self, docs: list[str], k1=1.5, b=0.75):
        self.k1, self.b = k1, b
        self.docs = [Counter(tokenize(d)) for d in docs]
        self.lens = [sum(d.values()) for d in self.docs]
        self.avg = (sum(self.lens) / len(self.lens)) if self.lens else 0
        df = Counter()
        for d in self.docs:
            df.update(d.keys())
        n = len(self.docs)
        self.idf = {w: math.log(1 + (n - f + 0.5) / (f + 0.5)) for w, f in df.items()}

    def scores(self, query: str) -> list[float]:
        q = tokenize(query)
        out = []
        for d, length in zip(self.docs, self.lens):
            s = 0.0
            for w in q:
                f = d.get(w)
                if f:
                    s += self.idf[w] * f * (self.k1 + 1) / (f + self.k1 * (1 - self.b + self.b * length / (self.avg or 1)))
            out.append(s)
        return out


_bm25_cache: dict[str, BM25] = {}


def _bm25(book) -> BM25:
    key = f"{book['id']}:{len(book['chunks'])}"
    if key not in _bm25_cache:
        _bm25_cache[key] = BM25([c["text"] for c in book["chunks"]])
    return _bm25_cache[key]


def search(book: dict, query: str, k: int = 6, chapter: int | None = None, query_vec=None) -> list[dict]:
    chunks = book["chunks"]
    allowed = [i for i, c in enumerate(chunks) if chapter is None or c["chapter"] == chapter]
    if not allowed:
        return []
    bm = _bm25(book).scores(query)
    ranked_lists = [sorted(allowed, key=lambda i: bm[i], reverse=True)]
    if query_vec is not None and book.get("embeddings"):
        m = np.asarray(book["embeddings"], dtype=np.float32)
        q = np.asarray(query_vec, dtype=np.float32)
        sims = m @ q / (np.linalg.norm(m, axis=1) * np.linalg.norm(q) + 1e-9)
        ranked_lists.append(sorted(allowed, key=lambda i: sims[i], reverse=True))
    elif max(bm[i] for i in allowed) == 0:
        return []  # no keyword overlap at all and no semantic search
    fused = Counter()
    for ranked in ranked_lists:
        for rank, i in enumerate(ranked[:50]):
            fused[i] += 1 / (60 + rank)
    top = [i for i, _ in fused.most_common(k)]
    return [chunks[i] | {"score": round(fused[i], 5)} for i in top]

def extractive_summary(text: str, n_sentences: int = 8) -> list[str]:
    """Pick the sentences that use the text's most important words."""
    from .books import split_sentences
    sents = [s for s in split_sentences(text)
             if 6 <= len(s.split()) <= 60 and re.search(r"[.!?…”\"']$", s)]
    if len(sents) <= n_sentences:
        return sents
    freq = Counter(tokenize(text))
    top = max(freq.values()) if freq else 1
    scored = []
    for i, s in enumerate(sents):
        words = tokenize(s)
        if not words:
            continue
        score = sum(freq[w] / top for w in words) / (len(words) ** 0.6)
        score *= 1.15 if i < len(sents) * 0.1 else 1.0  # openings matter
        scored.append((score, i))
    # spread picks across the text: best sentence from each slice
    picked = set()
    slices = max(1, n_sentences)
    size = len(sents) / slices
    for s in range(slices):
        lo, hi = int(s * size), int((s + 1) * size)
        best = max(((sc, i) for sc, i in scored if lo <= i < hi), default=None)
        if best:
            picked.add(best[1])
    return [sents[i] for i in sorted(picked)]


def best_sentences(passages: list[str], query: str, n: int = 4) -> list[str]:
    from .books import split_sentences
    q = set(tokenize(query))
    cands = []
    for p in passages:
        for s in split_sentences(p):
            overlap = len(q & set(tokenize(s)))
            if overlap:
                cands.append((overlap / (len(s.split()) ** 0.3), s))
    cands.sort(reverse=True)
    seen, out = set(), []
    for _, s in cands:
        if s not in seen:
            out.append(s)
            seen.add(s)
        if len(out) == n:
            break
    return out

"""One place for every OpenAI call: retries, friendly errors, token + cost logging."""
import time

import openai
from openai import OpenAI

from . import config, db
from .tokens import count_messages, count_tokens, estimate_cost


class LLMError(Exception):
    """An error with a status code, a short code, and a hint the UI can show."""

    def __init__(self, status: int, code: str, message: str, hint: str = ""):
        super().__init__(message)
        self.status, self.code, self.message, self.hint = status, code, message, hint

    def to_dict(self):
        return {"error": {"code": self.code, "message": self.message, "hint": self.hint}}


class Meter:
    """Adds up tokens and cost across the calls made for one user request."""

    def __init__(self):
        self.prompt_tokens = self.completion_tokens = self.calls = 0
        self.cost = 0.0
        self.started = time.time()

    def add(self, pt, ct, cost):
        self.prompt_tokens += pt
        self.completion_tokens += ct
        self.cost += cost
        self.calls += 1

    def to_dict(self):
        return {
            "prompt_tokens": self.prompt_tokens,
            "completion_tokens": self.completion_tokens,
            "total_tokens": self.prompt_tokens + self.completion_tokens,
            "cost_usd": round(self.cost, 6),
            "calls": self.calls,
            "latency_ms": int((time.time() - self.started) * 1000),
        }


def resolve_key(header_key: str | None) -> str:
    """A key typed in the UI (sent as X-OpenAI-Key) wins over the server's key."""
    return (header_key or "").strip() or config.OPENAI_API_KEY


def resolve_model(requested: str | None) -> str:

def map_error(e: Exception) -> LLMError:
    msg = str(e)
    if isinstance(e, openai.AuthenticationError):
        return LLMError(401, "invalid_api_key", "OpenAI rejected the API key.",
                        "Open Settings and paste a valid key (starts with sk-).")
    if isinstance(e, openai.PermissionDeniedError):
        return LLMError(403, "permission_denied", "This key can't use that model.",
                        "Pick another model in Settings.")
    if isinstance(e, openai.RateLimitError):
        if "insufficient_quota" in msg or "quota" in msg.lower():
            return LLMError(402, "quota_exceeded", "Your OpenAI account is out of credit.",
                            "Add billing at platform.openai.com, or keep using Demo mode.")
        return LLMError(429, "rate_limited", "Too many requests to OpenAI right now.",
                        "Wait a few seconds and try again.")
    if isinstance(e, openai.BadRequestError):
        if "context_length" in msg or "maximum context" in msg:
            return LLMError(400, "context_too_long", "The text is too long for the model.",
                            "Pick one chapter, or paste a shorter piece.")
        return LLMError(400, "bad_request", f"OpenAI said the request was invalid: {msg[:200]}")
    if isinstance(e, openai.NotFoundError):
        return LLMError(404, "model_not_found", "That model doesn't exist for this key.",
                        "Choose another model in Settings or set OPENAI_MODEL.")
    if isinstance(e, openai.APITimeoutError):
        return LLMError(504, "timeout", "OpenAI took too long to answer.", "Try again, or use shorter input.")
    if isinstance(e, openai.APIConnectionError):
        return LLMError(503, "network", "Couldn't reach OpenAI.", "Check the internet connection.")
    if isinstance(e, openai.APIStatusError):
        return LLMError(502, "upstream_error", "OpenAI had a server error.", "Try again in a moment.")
    return LLMError(500, "llm_error", f"Unexpected error: {msg[:200]}")


def _check_budget():
    if config.DAILY_BUDGET_USD > 0 and db.cost_today() >= config.DAILY_BUDGET_USD:
        raise LLMError(429, "budget_exceeded",
                       f"Today's spending limit (${config.DAILY_BUDGET_USD:.2f}) is reached.",
                       "Raise DAILY_BUDGET_USD or wait until tomorrow.")


def _client(key: str) -> OpenAI:
    return OpenAI(api_key=key, timeout=config.REQUEST_TIMEOUT, max_retries=config.MAX_RETRIES)


def _create(client, **kwargs):
    """Some newer models reject custom temperature; retry once without it."""
    try:
        return client.chat.completions.create(**kwargs)
    except openai.BadRequestError as e:
        if "temperature" in str(e) and "temperature" in kwargs:
            kwargs.pop("temperature")
            return client.chat.completions.create(**kwargs)
        raise


def chat(messages, *, key, feature, meter=None, model=None, temperature=0.4, max_tokens=1200) -> str:
    _check_budget()
    model = resolve_model(model)
    t0 = time.time()
    try:
        r = _create(_client(key), model=model, messages=messages,
                    temperature=temperature, max_completion_tokens=max_tokens)
    except openai.OpenAIError as e:
        err = map_error(e)
        db.log_usage(feature, model, 0, 0, 0, int((time.time() - t0) * 1000), ok=False, error=err.code)
        raise err from e
    text = r.choices[0].message.content or ""
    pt = r.usage.prompt_tokens if r.usage else count_messages(messages, model)
    ct = r.usage.completion_tokens if r.usage else count_tokens(text, model)
    cost = estimate_cost(model, pt, ct)
    db.log_usage(feature, model, pt, ct, cost, int((time.time() - t0) * 1000))
    if meter:
        meter.add(pt, ct, cost)
    if r.choices[0].finish_reason == "length":
        text += "\n\n_(Stopped at the length limit. Ask for a shorter output or say “continue”.)_"
    return text


def chat_stream(messages, *, key, feature, meter=None, model=None, temperature=0.4, max_tokens=1200):
    """Yields text pieces as they arrive. Logs usage at the end."""
    _check_budget()
    model = resolve_model(model)
    t0 = time.time()
    parts, usage = [], None
    try:
        stream = _create(_client(key), model=model, messages=messages, temperature=temperature,
                         max_completion_tokens=max_tokens, stream=True,
                         stream_options={"include_usage": True})
        for chunk in stream:
            if chunk.usage:
                usage = chunk.usage
            if chunk.choices and chunk.choices[0].delta and chunk.choices[0].delta.content:
                piece = chunk.choices[0].delta.content
                parts.append(piece)
                yield piece
    except openai.OpenAIError as e:
        err = map_error(e)
        db.log_usage(feature, model, 0, 0, 0, int((time.time() - t0) * 1000), ok=False, error=err.code)
        raise err from e
    text = "".join(parts)
    pt = usage.prompt_tokens if usage else count_messages(messages, model)
    ct = usage.completion_tokens if usage else count_tokens(text, model)
    cost = estimate_cost(model, pt, ct)
    db.log_usage(feature, model, pt, ct, cost, int((time.time() - t0) * 1000))
    if meter:
        meter.add(pt, ct, cost)


def embed(texts: list[str], *, key, feature="embeddings", meter=None) -> list[list[float]]:
    _check_budget()
    model = config.EMBED_MODEL
    out = []
    client = _client(key)
    for i in range(0, len(texts), 96):
        batch = texts[i:i + 96]
        t0 = time.time()
        try:
            r = client.embeddings.create(model=model, input=batch)
        except openai.OpenAIError as e:
            err = map_error(e)
            db.log_usage(feature, model, 0, 0, 0, int((time.time() - t0) * 1000), ok=False, error=err.code)
            raise err from e
        out.extend(d.embedding for d in r.data)
        pt = r.usage.prompt_tokens if r.usage else sum(count_tokens(t) for t in batch)
        cost = estimate_cost(model, pt)
        db.log_usage(feature, model, pt, 0, cost, int((time.time() - t0) * 1000))
        if meter:
            meter.add(pt, 0, cost)
    return out

"""Token counting and cost estimates.

Uses tiktoken (the same tokenizer OpenAI uses). If it's missing or can't load its
vocabulary file (no internet), falls back to the rule of thumb: 1 token ~ 4 characters.
"""
from . import config

try:
    import tiktoken
except ImportError:
    tiktoken = None

_encoders = {}
_tiktoken_broken = False


def _encoder(model: str):
    global _tiktoken_broken
    if tiktoken is None or _tiktoken_broken:
        return None
    if model not in _encoders:
        try:
            try:
                _encoders[model] = tiktoken.encoding_for_model(model)
            except KeyError:
                _encoders[model] = tiktoken.get_encoding("o200k_base")
        except Exception:
            _tiktoken_broken = True
            return None
    return _encoders[model]


def count_tokens(text: str, model: str | None = None) -> int:
    if not text:
        return 0
    enc = _encoder(model or config.CHAT_MODEL)
    if enc is not None:
        return len(enc.encode(text, disallowed_special=()))
    return max(1, len(text) // 4)


def count_messages(messages: list[dict], model: str | None = None) -> int:
    # ~4 extra tokens per message for role/formatting, +2 for the reply primer
    return sum(count_tokens(m.get("content") or "", model) + 4 for m in messages) + 2


def truncate(text: str, max_tokens: int, model: str | None = None) -> str:
    enc = _encoder(model or config.CHAT_MODEL)
    if enc is not None:
        ids = enc.encode(text, disallowed_special=())
        return text if len(ids) <= max_tokens else enc.decode(ids[:max_tokens])
    return text[: max_tokens * 4]


def estimate_cost(model: str, prompt_tokens: int, completion_tokens: int = 0) -> float:
    price_in, price_out = config.PRICES.get(model, config.PRICES["gpt-4o-mini"])
    return (prompt_tokens * price_in + completion_tokens * price_out) / 1_000_000


def tokenizer_name() -> str:
    return "tiktoken" if _encoder(config.CHAT_MODEL) is not None else "approx (chars/4)"

"""SQLite storage for usage tracking and user feedback.

On Vercel the file lives in /tmp and is lost when the instance restarts, so the
frontend also keeps its own copy in the browser (see static/app.js).

import sqlite3
import time
from contextlib import contextmanager

from . import config

DB_PATH = config.DATA_DIR / "studio.db"

CREATE TABLE IF NOT EXISTS usage (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ts REAL NOT NULL,
    feature TEXT NOT NULL,
    model TEXT NOT NULL,
    prompt_tokens INTEGER DEFAULT 0,
    completion_tokens INTEGER DEFAULT 0,
    cost REAL DEFAULT 0,
    latency_ms INTEGER DEFAULT 0,
    ok INTEGER DEFAULT 1,
    error TEXT
);
CREATE TABLE IF NOT EXISTS feedback (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ts REAL NOT NULL,
    feature TEXT NOT NULL,
    prompt_version TEXT,
    rating INTEGER NOT NULL,
    comment TEXT,
    output_id TEXT,
    output_excerpt TEXT
);

@contextmanager
def connect():

    conn.row_factory = sqlite3.Row

    start = time.time() - (time.time() % 86400)
    with connect() as c:
        row = c.execute("SELECT COALESCE(SUM(cost),0) FROM usage WHERE ts >= ?", (start,)).fetchone()
    return float(row[0])

7. app/services.py: summarise, chat, generate

book_genai_studio/app/services.py
What it does

The brain of the app. Each feature has a live path (OpenAI) and a demo path (offline).

The logic, step by step
  1. Summarise: if there's no key, it uses extractive_summary plus keywords. If the text fits (under 12,000 tokens), it makes one call.
  2. If the text is too long, it uses map-reduce: _notes() first writes short notes for each piece (map), then a final call combines the notes (reduce). Whole-book summaries run one chapter at a time, and the notes are cached.
  3. Chat (chat_events) is a generator that streams Server-Sent Events: meta (the sources), many delta events (pieces of text), then done.
  4. Chat step 1: if there are earlier questions, the follow-up is rewritten into a standalone search question.
  5. Chat step 2: search() finds 6 passages, which are tagged [S1]…[S6] with their chapter and page.
  6. Chat step 3: the passages go into the system prompt, and the model streams an answer that cites [S1], [S2]…
  7. Generate: _material() collects text (from the book, a chapter, a topic search, or pasted text), then fills the GENERATE prompt for the chosen type: quiz, flashcards, notes, blog and so on.

SINGLE_CALL_LIMIT = 12000   # tokens we send in one summarize call
WINDOW = 6000               # piece size for map-reduce
DEMO_NOTE = "\n\n> **Demo mode** – made offline without AI. Add an OpenAI key in Settings for real AI output."


def _out_id():
    return uuid.uuid4().hex[:10]


def _windows(text: str, size: int = WINDOW) -> list[str]:
    """Split text into pieces of about `size` tokens, on paragraph breaks."""
    out, buf = [], []
    buf_tokens = 0
    for para in re.split(r"\n\s*\n", text):
        t = count_tokens(para)
        if buf and buf_tokens + t > size:
            out.append("\n\n".join(buf))
            buf, buf_tokens = [], 0
        buf.append(truncate(para, size))
        buf_tokens += min(t, size)
    if buf:
        out.append("\n\n".join(buf))
    return out

def _notes(text, title, part, key, model, meter) -> str:
    """Dense notes for one piece of a book (map step)."""
    if not key:
        return " ".join(extractive_summary(text, 6))
    pieces = _windows(text) if count_tokens(text) > SINGLE_CALL_LIMIT else [text]

    def one(i_piece):
        i, piece = i_piece
        label = part if len(pieces) == 1 else f"{part}, piece {i + 1}/{len(pieces)}"
        return llm.chat([{"role": "user", "content": prompts.NOTES_PROMPT.format(title=title, part=label, text=piece)}],
                        key=key, feature="summarize", meter=meter, model=model, temperature=0.2, max_tokens=500)

    with ThreadPoolExecutor(max_workers=4) as pool:
        return "\n\n".join(pool.map(one, enumerate(pieces)))


def summarize(*, text, title, scope, length, style, version, key, model, book=None, chapter=None, notes=None):
    """scope: text | chapter | notes (map step for one chapter) | combine (reduce step)."""
    meter = Meter()
    words = prompts.LENGTHS.get(length, 300)
    style_desc = prompts.STYLES.get(style, prompts.STYLES["bullets"])
    version = prompts.pick_version("summarize", version)
    p = prompts.SUMMARIZE[version]

    if scope == "notes":
        ch = book["chapters"][chapter]
        cache_key = f"notes:{chapter}"
        cached = (book.get("cache") or {}).get(cache_key) if key else None
        out = cached or _notes(text, title, ch["title"], key, model, meter)
        if key and not cached:
            book.setdefault("cache", {})[cache_key] = out
        return {"output": out, "cached": bool(cached), "usage": meter.to_dict(), "mode": "live" if key else "demo"}

    if scope == "combine":
        joined = "\n\n".join(f"## {n['title']}\n{n['notes']}" for n in notes)
        text = joined

    tokens_in = count_tokens(text)
    if not key:
        n = {"short": 4, "medium": 8, "detailed": 15}.get(length, 8)
        sents = extractive_summary(text, n)
        kw = ", ".join(keywords(text, 8))
        out = f"**Key ideas:** {kw}\n\n" + "\n".join(f"- {s}" for s in sents) + DEMO_NOTE
        return {"output": out, "output_id": _out_id(), "prompt_version": "demo", "usage": meter.to_dict(),
                "tokens_in": tokens_in, "mode": "demo"}

    if scope == "combine":
        msgs = [{"role": "system", "content": prompts.SUMMARIZE["v2"]["system"]},
                {"role": "user", "content": prompts.COMBINE_PROMPT.format(
                    title=title, words=words, style=style_desc, text=truncate(text, SINGLE_CALL_LIMIT * 2))}]
    else:
        if tokens_in > SINGLE_CALL_LIMIT:  # long chapter / long paste: map-reduce
            text = _notes(text, title, "this section", key, model, meter)
        msgs = [{"role": "system", "content": p["system"]},
                {"role": "user", "content": p["user"].format(words=words, style=style_desc, text=text)}]
    out = llm.chat(msgs, key=key, feature="summarize", meter=meter, model=model,
                   temperature=0.3, max_tokens=int(words * 2.2) + 200)
    return {"output": out, "output_id": _out_id(), "prompt_version": version, "usage": meter.to_dict(),
            "tokens_in": tokens_in, "mode": "live"}

def _sse(event: str, data) -> str:
    return f"event: {event}\ndata: {json.dumps(data)}\n\n"


def _trim_history(messages: list[dict], budget: int = 3000) -> list[dict]:
    """Keep the newest messages that fit in the token budget."""
    kept, used = [], 0
    for m in reversed(messages):
        t = count_tokens(m["content"])
        if kept and used + t > budget:
            break
        kept.append({"role": m["role"], "content": truncate(m["content"], budget)})
        used += t
    return kept[::-1]


def _source_label(book, c):
    ch = book["chapters"][c["chapter"]]["title"]
    return f"{ch}" + (f", p. {c['page']}" if c.get("page") else "")


def chat_events(*, book, messages, chapter, version, key, model):
    """Generator of Server-Sent Events: meta -> delta* -> done (or error)."""
    meter = Meter()
    out_id = _out_id()
    try:
        history = _trim_history(messages)
        question = history[-1]["content"]
        if not book:
            if not key:
                yield _sse("meta", {"sources": [], "prompt_version": "demo", "output_id": out_id})
                msg = ("I'm in **Demo mode**, so I can only answer from an uploaded book. Upload a book and pick "
                       "it at the top, or add an OpenAI key in Settings to chat about anything.")
                yield _sse("delta", {"t": msg})
                yield _sse("done", {"usage": meter.to_dict(), "mode": "demo"})
                return
            yield _sse("meta", {"sources": [], "prompt_version": "general", "output_id": out_id})
            msgs = [{"role": "system", "content": prompts.GENERAL_CHAT_SYSTEM}] + history
            for piece in llm.chat_stream(msgs, key=key, feature="chat", meter=meter, model=model):
                yield _sse("delta", {"t": piece})
            yield _sse("done", {"usage": meter.to_dict(), "mode": "live"})
            return

        # 1. Follow-up questions ("what did she do next?") -> standalone search question
        search_q = question
        user_turns = [m for m in history if m["role"] == "user"]
        if key and len(user_turns) > 1:
            hist = "\n".join(f"{m['role']}: {m['content'][:400]}" for m in history[-6:-1])
            search_q = llm.chat([{"role": "user", "content": prompts.CONDENSE_PROMPT.format(history=hist, question=question)}],
                                key=key, feature="qa_rewrite", meter=meter, model=model, temperature=0, max_tokens=80).strip()

        # 2. Retrieve passages (keyword + semantic when available)
        qvec = None
        if key and book.get("embeddings"):
            qvec = llm.embed([search_q], key=key, feature="embeddings", meter=meter)[0]
        hits = search(book, search_q, k=6, chapter=chapter, query_vec=qvec)
        sources = [{"tag": f"S{i + 1}", "label": _source_label(book, c), "chapter": c["chapter"],
                    "page": c.get("page"), "text": c["text"][:1200]} for i, c in enumerate(hits)]
        version = prompts.pick_version("qa", version)
        yield _sse("meta", {"sources": sources, "prompt_version": version if key else "demo",
                            "output_id": out_id, "search_query": search_q})

        # 3. Answer
        if not key:
            if not hits:
                ans = "I couldn't find anything about that in the book. Try other words, like a character's name."
            else:
                sents = best_sentences([h["text"] for h in hits[:4]], search_q, 4)
                ans = "Here's what the book says:\n\n" + "\n".join(
                    f"- {s} [S{next((i + 1 for i, h in enumerate(hits) if s[:40] in h['text'].replace(chr(10), ' ')), 1)}]"
                    for s in sents) + DEMO_NOTE
            for w in re.findall(r"\S+\s*", ans):
                yield _sse("delta", {"t": w})
            yield _sse("done", {"usage": meter.to_dict(), "mode": "demo"})
            return

        context = "\n\n".join(f"[{s['tag']}] ({s['label']})\n{h['text']}" for s, h in zip(sources, hits)) \
            or "(no matching passages found)"
        system = prompts.QA[version]["system"].format(title=book["title"], context=context)
        msgs = [{"role": "system", "content": system}] + history
        for piece in llm.chat_stream(msgs, key=key, feature="qa", meter=meter, model=model, temperature=0.2):
            yield _sse("delta", {"t": piece})
        yield _sse("done", {"usage": meter.to_dict(), "mode": "live"})
    except LLMError as e:
        yield _sse("error", e.to_dict()["error"])
    except Exception as e:  # never leave the browser hanging
        yield _sse("error", {"code": "server_error", "message": str(e)[:200], "hint": "Try again."})

def _material(book, chapter, topic, text, budget=9000) -> str:
    if text:
        return truncate(text, budget)
    if chapter is not None:
        ch = book["chapters"][chapter]
        return truncate(book["text"][ch["start"]:ch["end"]], budget)
    if topic:
        hits = search(book, topic, k=12)
        if hits:
            return "\n\n".join(h["text"] for h in hits)
    # whole book: evenly spaced passages from start to end
    chunks = book["chunks"]
    take = max(1, min(len(chunks), budget // 450))
    step = len(chunks) / take
    return "\n\n".join(chunks[int(i * step)]["text"] for i in range(take))


def _demo_content(kind: str, material: str) -> str:
    sents = extractive_summary(material, 15)
    kw = keywords(material, 20)
    if kind == "quiz":
        lines, n = ["## Quiz (fill in the blank)"], 0
        for s in sents:
            word = next((w for w in kw if re.search(rf"\b{re.escape(w)}\b", s, re.I)), None)
            if not word:
                continue
            n += 1
            options = random.sample([w for w in kw if w != word], k=min(3, len(kw) - 1)) + [word]
            random.shuffle(options)
            q = re.sub(rf"\b{re.escape(word)}\b", "_____", s, count=1, flags=re.I)
            lines.append(f"**{n}. {q}**\n" + "\n".join(f"- {chr(65 + i)}) {o}" for i, o in enumerate(options))
                         + f"\n\n*Answer: {word}*\n")
            if n == 8:
                break
        return "\n".join(lines)
    if kind == "flashcards":
        rows, used = ["| Front | Back |", "|---|---|"], set()
        for s in sents[:12]:
            word = next((w for w in kw if w not in used and re.search(rf"\b{re.escape(w)}\b", s, re.I)), None)
            if word:
                used.add(word)
                rows.append(f"| What does the text say about **{word}**? | {s.replace('|', '/')} |")
        return "\n".join(rows)
    if kind == "timeline":
        return "| # | Event |\n|---|---|\n" + "\n".join(f"| {i + 1} | {s.replace('|', '/')} |" for i, s in enumerate(sents[:10]))
    return (f"## Key themes\n{', '.join(kw[:10])}\n\n## Main points\n" + "\n".join(f"- {s}" for s in sents[:10]))


def generate(*, book, chapter, topic, text, content_type, tone, audience, extra, version, key, model):
    meter = Meter()
    if not text and not book:
        if not topic:
            raise LLMError(400, "no_input", "Pick a book, paste text, or type a topic.")
        if not key:
            raise LLMError(400, "needs_key", "Demo mode can only create content from a book or pasted text.",
                           "Upload a book, paste text, or add an OpenAI key in Settings.")
    material = _material(book, chapter, topic, text) if (book or text) else f"Topic: {topic}"
    title = book["title"] if book else "the provided material"
    if not key:
        out = _demo_content(content_type, material) + DEMO_NOTE
        return {"output": out, "output_id": _out_id(), "prompt_version": "demo", "usage": meter.to_dict(),
                "tokens_in": count_tokens(material), "mode": "demo"}
    version = prompts.pick_version("generate", version)
    p = prompts.GENERATE[version]
    spec = prompts.CONTENT_TYPES.get(content_type, prompts.CONTENT_TYPES["custom"])

                            extra=extra or "none", material=material, title=title)
    out = llm.chat([{"role": "system", "content": p["system"]}, {"role": "user", "content": user}],

The logic, step by step

  2. idb stores processed books in the browser's IndexedDB, so they survive server restarts.

async function api(path, { method = "GET", body, form, stream = false, signal } = {}, retry = true) {
  const headers = {};
  if (state.settings.apiKey) headers["X-OpenAI-Key"] = state.settings.apiKey;
  let payload;
  if (form) payload = form;
  else if (body !== undefined) { headers["Content-Type"] = "application/json"; payload = JSON.stringify(body); }
  let res;
  try { res = await fetch(path, { method, headers, body: payload, signal }); }
  catch (e) {
    if (e.name === "AbortError") throw e;
    throw new AppError("network", "Can't reach the server.", "Check your connection, or that the server is running.");
  }
  if (!res.ok) {
    let data = {};
    try { data = await res.json(); } catch { /* not JSON */ }
    const err = data.error || {};
    if (Array.isArray(data.detail)) {   // FastAPI validation error
      const d = data.detail[0];
      err.code = "invalid_input"; err.message = `${(d.loc || []).slice(1).join(".")}: ${d.msg}`;
    }
    if (err.code === "book_not_found" && retry) {
      const bookId = body?.book_id || path.match(/\/api\/books\/([a-f0-9]+)/)?.[1] || body?.summarize?.book_id || body?.generate?.book_id;
      if (bookId && await restoreBook(bookId)) return api(path, { method, body, form, stream, signal }, false);
    }
    if (res.status === 413 && !err.code) { err.code = "too_large"; err.message = "That's too big for the server."; err.hint = "Vercel's free plan accepts up to ~4.5 MB per request."; }
    throw new AppError(err.code || `http_${res.status}`, err.message || `Server error (${res.status})`, err.hint || "");
  }
  return stream ? res : res.json();
}

      book_id: c.bookId || null, chapter: chapterVal === "" ? null : +chapterVal, model: model(),
      prompt_version: "auto", messages: c.messages.slice(0, -1).filter((m) => !m.error).map(({ role, content }) => ({ role, content })) } });
    const reader = res.body.getReader(), dec = new TextDecoder();
    let buf = "", lastPaint = 0;
    const paint = (force) => {
      if (!force && Date.now() - lastPaint < 50) return;
      lastPaint = Date.now();
      const el = bodyEl(); $(".body", el).innerHTML = md(msg.content);
      const box = $("#messages"); box.scrollTop = box.scrollHeight;
    };
    for (;;) {
      const { value, done } = await reader.read();
      if (done) break;
      buf += dec.decode(value, { stream: true });
      let cut;
      while ((cut = buf.indexOf("\n\n")) >= 0) {
        const raw = buf.slice(0, cut); buf = buf.slice(cut + 2);
        const ev = raw.match(/^event: (.*)$/m)?.[1], data = JSON.parse(raw.match(/^data: (.*)$/m)?.[1] || "{}");
        if (ev === "meta") { Object.assign(msg, { sources: data.sources, prompt_version: data.prompt_version, output_id: data.output_id }); }
        else if (ev === "delta") { msg.content += data.t; paint(); }
        else if (ev === "done") { msg.usage = data.usage; msg.mode = data.mode; recordUsage("qa", data.usage); }
        else if (ev === "error") { throw new AppError(data.code, data.message, data.hint); }
      }
    }
  } catch (e) {
    if (e.name === "AbortError") msg.content += "\n\n_(stopped)_";
    else { msg.error = true; msg.content = (msg.content ? msg.content + "\n\n" : "") + `⚠️ **${e.message}** ${e.hint || ""}`; showError(e); }
  } finally {
    state.chatAbort = null;
    $("#sendBtn").classList.remove("hidden"); $("#stopBtn").classList.add("hidden");
    saveConvos(); renderMessages();
  }
}

def test_live_followup_question_is_rewritten(book, fake_openai):
    msgs = [{"role": "user", "content": "Who is Elias?"}, {"role": "assistant", "content": "The keeper."},
            {"role": "user", "content": "What did he promise?"}]
    r = client.post("/api/chat", json={"book_id": book["book"]["id"], "messages": msgs},
                    headers={"X-OpenAI-Key": "sk-test"})
    events = sse(r.text)
    assert [f for f, _ in fake_openai] == ["qa_rewrite", "qa"]
    text = "".join(d["t"] for e, d in events if e == "delta")
    assert text == "The answer is Adaeze [S1]."
    assert events[-1][1]["usage"]["calls"] == 2
