Day 19: Four agent architectures with LangGraph

Each agent is a graph: nodes are steps (often one LLM call each), edges say what runs next, and a shared State dictionary carries the data between them.

Programs on this page
  1. Router agent: pick the right specialist
  2. Reflection agent: write, critique, revise
  3. Supervisor agent: a boss directs workers
  4. Human-in-the-loop agent: nothing is sent until you approve

1. Router agent: pick the right specialist

router_agent/agent.py · router_agent/tools.py
What it does

A customer message is sorted into billing, tech or general, and that specialist answers it.

How to run
cd day_19\router_agent python agent.py
The logic, step by step
  1. router_node asks the LLM for ONE word, using a prompt with examples. The code then searches that word for "billing" or "tech", and anything else becomes general. That safe fallback matters, because small models add extra words.
  2. add_conditional_edges("router", lambda s: s["route"], {...}) is the if/else that chooses the next node.
  3. The billing and tech specialists are their own small ReAct agents with their own tools. The tools use fake local data (INVOICES, KNOWLEDGE_BASE), so the demo works offline.
  4. Each specialist writes answer into the state, then the graph ends.
Flow
START → router ─┬─ billing (agent + lookup_invoice) ─┐
                 ├─ tech    (agent + search_kb)      ─┼─▶ END
                 └─ general (plain LLM)              ─┘

from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import create_react_agent
from langchain_core.messages import HumanMessage, SystemMessage
from tools import lookup_invoice, search_kb

OLLAMA_MODEL = "llama3.2"
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

ROUTER_PROMPT = """You are a router for a customer-support team.
Read the customer message and answer with EXACTLY ONE word:
  billing  -> invoices, payments, refunds, charges
  tech     -> bugs, crashes, wifi, passwords, how-to problems
  general  -> anything else (hours, greetings, feedback)

Examples:
  "Is invoice 1002 paid?"          -> billing
  "I was charged twice"            -> billing
  "The app crashes on startup"     -> tech
  "I forgot my password"           -> tech
  "What time do you open?"         -> general
  "Hello!"                         -> general

Answer with just the one word."""

BILLING_PROMPT = "You are a friendly billing specialist. Use lookup_invoice when an invoice number is given."
TECH_PROMPT = "You are a patient tech-support engineer. Use search_kb to find fixes, then explain step by step."
GENERAL_PROMPT = "You are a polite front-desk assistant. Keep answers short and helpful."

# Each specialist is its own small agent with its own tools.
billing_agent = create_react_agent(llm, [lookup_invoice])
tech_agent = create_react_agent(llm, [search_kb])


# ---------- State: what flows through the graph ----------
class State(TypedDict):
    question: str
    route: str
    answer: str


# ---------- Nodes ----------
def router_node(state: State):
    reply = llm.invoke([SystemMessage(content=ROUTER_PROMPT), HumanMessage(content=state["question"])])
    word = reply.content.strip().lower()
    route = next((r for r in ("billing", "tech") if r in word), "general")  # safe fallback
    print(f"  [router] -> {route}")
    return {"route": route}


def billing_node(state: State):
    result = billing_agent.invoke({"messages": [SystemMessage(content=BILLING_PROMPT), HumanMessage(content=state["question"])]})
    return {"answer": result["messages"][-1].content}


def tech_node(state: State):
    result = tech_agent.invoke({"messages": [SystemMessage(content=TECH_PROMPT), HumanMessage(content=state["question"])]})
    return {"answer": result["messages"][-1].content}


def general_node(state: State):
    reply = llm.invoke([SystemMessage(content=GENERAL_PROMPT), HumanMessage(content=state["question"])])
    return {"answer": reply.content}


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("router", router_node)
    g.add_node("billing", billing_node)
    g.add_node("tech", tech_node)
    g.add_node("general", general_node)

    g.add_edge(START, "router")
    # The "if/else" of agents: pick the next node from state["route"]
    g.add_conditional_edges("router", lambda s: s["route"],
                            {"billing": "billing", "tech": "tech", "general": "general"})
    for name in ("billing", "tech", "general"):
        g.add_edge(name, END)
    return g.compile()


def ask(agent, user_input):
    return agent.invoke({"question": user_input})["answer"]


def main():
    print("=" * 50)
    print("  🚦  Router Agent  (Ollama + LangGraph)")
    print("=" * 50)
    print(f"  Model: {OLLAMA_MODEL}  |  Routes: billing / tech / general")
    print("  Try: 'Is invoice 1002 paid?' or 'My wifi keeps dropping' or 'What are your hours?'\n")
    agent = build_agent()
    while True:
        try:
            user_input = input("You: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if not user_input: continue
        if user_input.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print("\nAgent:", ask(agent, user_input), "\n")


if __name__ == "__main__":
    main()

from langchain_core.tools import tool

# Fake "databases" so the demo works offline — no API key needed.
INVOICES = {
    "1001": "Invoice 1001: $49.00, paid on 2026-10-02.",
    "1002": "Invoice 1002: $120.00, OVERDUE since 2026-10-20.",
}

KNOWLEDGE_BASE = {
    "wifi": "Restart the router, wait 30 seconds, then reconnect. Still stuck? Forget the network and rejoin.",
    "password": "Click 'Forgot password' on the login page. A reset link is emailed within 2 minutes.",
    "crash": "Update the app to the latest version, then clear the cache in Settings > Storage.",
}


@tool
def lookup_invoice(invoice_id: str) -> str:
    """Look up an invoice by its number (e.g. 1001) and return amount and payment status."""
    return INVOICES.get(invoice_id.strip(), f"No invoice found with id '{invoice_id}'.")


@tool
def search_kb(keyword: str) -> str:
    """Search the tech-support knowledge base for a keyword like wifi, password or crash."""
    for key, answer in KNOWLEDGE_BASE.items():
        if key in keyword.lower():
            return answer
    return "No article found. Ask the customer for more details."

2. Reflection agent: write, critique, revise

reflection_agent/agent.py · reflection_agent/tools.py
What it does

A writer drafts text, a critic reviews it, and a reviser improves it. This repeats until the critic approves, or after 3 rounds.

How to run
cd day_19\reflection_agent python agent.py
The logic, step by step
  1. write_node makes the first draft (at most 100 words). clean() removes intros like "Here is the revised text:".
  2. critique_node sends the request, the word count and the draft to a strict editor prompt. The reply is either exactly APPROVED, or 2-3 suggestions.
  3. Guard rail: if the draft is over 100 words but the model still says APPROVED, the code overrides it with "too long". The code enforces the rule, not the model.
  4. should_continue returns "done" if approved or after MAX_ROUNDS (3), otherwise "revise".
  5. The edge revise → critique creates the loop. MAX_ROUNDS guarantees it always stops.
Flow
START → write → critique ──APPROVED or 3 rounds──▶ END
                    ▲   │ suggestions
                    └ revise ◀┘

from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage
from tools import count_words

OLLAMA_MODEL = "llama3.2"
MAX_ROUNDS = 3      # safety limit so the loop always stops
MAX_WORDS = 100     # the writer, critic AND reviser all enforce this

llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.5)

WRITER_PROMPT = f"""You are a skilled writer. Write a short, clear piece (max {MAX_WORDS} words)
for the user's request. Return ONLY the text itself — no title, no intro like "Here is...". """

CRITIC_PROMPT = f"""You are a strict but fair editor. Review the draft against the request.
Check: clarity, tone, grammar, and whether it fully answers the request.
HARD RULE: the draft must be {MAX_WORDS} words or fewer. If it is longer, it is NOT approved —
tell the writer to cut it.
- If it is good enough AND within the limit, reply with exactly: APPROVED
- Otherwise reply with 2-3 short, specific suggestions (one line each)."""

REVISER_PROMPT = f"""You are a writer revising your draft.
Rewrite it applying the editor's feedback, but keep it to {MAX_WORDS} words or fewer
(cut words if needed — never make it longer).
Return ONLY the improved text. Do NOT start with "Here is the revised text" or any intro."""


# ---------- State ----------
class State(TypedDict):
    request: str
    draft: str
    feedback: str
    rounds: int


def clean(text: str) -> str:
    """Small models love to add 'Here is the revised text:' — drop that intro line."""
    lines = text.strip().splitlines()
    if lines and lines[0].lower().startswith("here") and lines[0].rstrip().endswith(":"):
        lines = lines[1:]
    return "\n".join(lines).strip()


def is_approved(feedback: str) -> bool:
    """Approved only if the reply STARTS with APPROVED (not suggestions that end with the word)."""
    return feedback.strip().upper().startswith("APPROVED")


# ---------- Nodes ----------
def write_node(state: State):
    reply = llm.invoke([SystemMessage(content=WRITER_PROMPT), HumanMessage(content=state["request"])])
    draft = clean(reply.content)
    print(f"\n  ✍️  [writer] first draft ({count_words.invoke(draft)})")
    return {"draft": draft, "rounds": 0}


def critique_node(state: State):
    n = len(state["draft"].split())
    msg = f"Request: {state['request']}\n\nWord count: {n} (limit {MAX_WORDS})\n\nDraft:\n{state['draft']}"
    reply = llm.invoke([SystemMessage(content=CRITIC_PROMPT), HumanMessage(content=msg)])
    feedback = reply.content.strip()
    if n > MAX_WORDS and is_approved(feedback):
        # Guard rail: a small model may approve anyway, but code enforces the rule.
        feedback = f"Too long ({n} words). Cut it to {MAX_WORDS} words or fewer."
    print("  🧐 [critic]")
    for line in feedback.splitlines():
        if line.strip():
            print(f"       {line.strip()}")
    return {"feedback": feedback}


def revise_node(state: State):
    msg = f"Request: {state['request']}\n\nDraft:\n{state['draft']}\n\nEditor feedback:\n{state['feedback']}"
    reply = llm.invoke([SystemMessage(content=REVISER_PROMPT), HumanMessage(content=msg)])
    draft = clean(reply.content)
    rounds = state["rounds"] + 1
    print(f"  🔁 [reviser] revision #{rounds} ({count_words.invoke(draft)})")
    return {"draft": draft, "rounds": rounds}


# ---------- Loop decision ----------
def should_continue(state: State):
    if is_approved(state["feedback"]):
        return "done"
    if state["rounds"] >= MAX_ROUNDS:
        return "done"
    return "revise"


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("write", write_node)
    g.add_node("critique", critique_node)
    g.add_node("revise", revise_node)

    g.add_edge(START, "write")
    g.add_edge("write", "critique")
    g.add_conditional_edges("critique", should_continue, {"revise": "revise", "done": END})
    g.add_edge("revise", "critique")  # <-- the loop: revise, then critique again
    return g.compile()


def ask(agent, user_input):
    return agent.invoke({"request": user_input})["draft"]


def main():
    print("=" * 50)
    print("  🪞  Reflection Agent  (Ollama + LangGraph)")
    print("=" * 50)
    print(f"  Model: {OLLAMA_MODEL}  |  Loop: write → critique → revise (max {MAX_ROUNDS}, {MAX_WORDS} words)")
    print("  Tell me WHAT to write, e.g. 'Write a short email asking my boss for Friday off'\n")
    agent = build_agent()
    while True:
        try:
            user_input = input("What should I write? (q to quit): ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if user_input.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        if len(user_input.split()) < 4:  # 'good', 'pls proceed' are not writing requests
            print("   ⚠️  Describe what to write in a few words, e.g. 'Write a poem about rain'.\n")
            continue
        print("\nFinal:", ask(agent, user_input), "\n")


if __name__ == "__main__":
    main()

@tool
def count_words(text: str) -> str:
    """Count the words in a piece of text and report whether it is short, medium or long."""
    n = len(text.split())
    size = "short" if n < 60 else "medium" if n < 150 else "long"

3. Supervisor agent: a boss directs workers

supervisor_agent/agent.py · supervisor_agent/tools.py
What it does

A supervisor decides who works next: the researcher (Wikipedia), the writer, or the proofreader. Each worker reports back.

How to run
cd day_19\supervisor_agent python agent.py
The logic, step by step
  1. progress(state) tells the supervisor what's done: notes yes/no, article yes/no.
  2. The supervisor LLM picks one word. Guard rails then enforce the order in code: no notes → researcher, no article → writer, not proofread → proofreader, otherwise FINISH. MAX_STEPS = 6 is a safety stop.
  3. The researcher is a ReAct agent with the wiki_summary tool (Wikipedia REST API), and it saves bullet notes.
  4. The writer turns the notes into an 80-word article. The proofreader polishes it and adds "(proofread ✔)" as a marker.
  5. Every worker has an edge back to the supervisor. That is the hub-and-spoke shape.
Flow
          ┌──▶ researcher ──┐
START → supervisor ──▶ writer ──────┤ (back to supervisor)
          └──▶ proofreader ─┘
   supervisor says FINISH → END

from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import create_react_agent
from langchain_core.messages import HumanMessage, SystemMessage
from tools import wiki_summary

OLLAMA_MODEL = "llama3.2"
MAX_STEPS = 6  # safety limit on supervisor decisions

llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

SUPERVISOR_PROMPT = """You are a supervisor managing a team of workers:
  researcher -> finds facts (Wikipedia)
  writer     -> turns the facts into a short article
  proofreader-> fixes grammar and polishes the article
Look at the progress so far and decide who works NEXT.
Typical order: researcher, then writer, then proofreader, then FINISH.
Answer with EXACTLY ONE word: researcher, writer, proofreader, or FINISH."""

RESEARCHER_PROMPT = "You are a researcher. Use wiki_summary to get facts about the topic. Report the key facts as bullet points."
WRITER_PROMPT = "You are a writer. Using the research notes, write a clear 80-word article. Return only the article."
PROOFREADER_PROMPT = "You are a proofreader. Fix grammar and polish this article without changing its meaning. Return only the final article."

researcher_agent = create_react_agent(llm, [wiki_summary])  # a worker WITH a tool


# ---------- State ----------
class State(TypedDict):
    task: str
    notes: str       # researcher output
    article: str     # writer / proofreader output
    next: str        # supervisor's decision
    steps: int


def progress(state: State) -> str:
    return (f"Task: {state['task']}\n"
            f"Research notes done: {'yes' if state['notes'] else 'no'}\n"
            f"Article written: {'yes' if state['article'] else 'no'}")


# ---------- Supervisor ----------
def supervisor_node(state: State):
    reply = llm.invoke([SystemMessage(content=SUPERVISOR_PROMPT), HumanMessage(content=progress(state))])
    word = reply.content.strip().lower()
    choice = next((w for w in ("researcher", "writer", "proofreader") if w in word), "FINISH")
    # Guard rails: small models sometimes pick a wrong step, so enforce the order.
    if not state["notes"]:
        choice = "researcher"
    elif not state["article"]:
        choice = "writer"
    elif "proofread ✔" in state["article"]:
        choice = "FINISH"
    else:
        choice = "proofreader"
    if state["steps"] >= MAX_STEPS:
        choice = "FINISH"
    print(f"  🧑‍💼 [supervisor] -> {choice}")
    return {"next": choice, "steps": state["steps"] + 1}


# ---------- Workers ----------
def researcher_node(state: State):
    result = researcher_agent.invoke({"messages": [SystemMessage(content=RESEARCHER_PROMPT), HumanMessage(content=state["task"])]})
    return {"notes": result["messages"][-1].content}


def writer_node(state: State):
    msg = f"Topic: {state['task']}\n\nResearch notes:\n{state['notes']}"
    reply = llm.invoke([SystemMessage(content=WRITER_PROMPT), HumanMessage(content=msg)])
    return {"article": reply.content}


def proofreader_node(state: State):
    reply = llm.invoke([SystemMessage(content=PROOFREADER_PROMPT), HumanMessage(content=state["article"])])
    return {"article": reply.content + "\n\n(proofread ✔)"}


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("supervisor", supervisor_node)
    g.add_node("researcher", researcher_node)
    g.add_node("writer", writer_node)
    g.add_node("proofreader", proofreader_node)

    g.add_edge(START, "supervisor")
    g.add_conditional_edges("supervisor", lambda s: s["next"],
                            {"researcher": "researcher", "writer": "writer",
                             "proofreader": "proofreader", "FINISH": END})
    # Every worker reports BACK to the supervisor (that's the hub-and-spoke shape).
    for worker in ("researcher", "writer", "proofreader"):
        g.add_edge(worker, "supervisor")
    return g.compile()


def ask(agent, user_input):
    result = agent.invoke({"task": user_input, "notes": "", "article": "", "next": "", "steps": 0})
    return result["article"]


def main():
    print("=" * 50)
    print("  🧑‍💼  Supervisor Agent  (Ollama + Wikipedia)")
    print("=" * 50)
    print(f"  Model: {OLLAMA_MODEL}  |  Workers: researcher, writer, proofreader")
    print("  Try: 'Black holes' or 'The Eiffel Tower'\n")
    agent = build_agent()
    while True:
        try:
            user_input = input("Topic: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if not user_input: continue
        if user_input.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print("\nArticle:", ask(agent, user_input), "\n")


if __name__ == "__main__":
    main()

import requests
from langchain_core.tools import tool


@tool
def wiki_summary(topic: str) -> str:
    """
    Get a short factual summary of a topic from Wikipedia.
    Uses the free Wikipedia REST API — no API key.
    """
    title = topic.strip().replace(" ", "_")
    r = requests.get(
        f"https://en.wikipedia.org/api/rest_v1/page/summary/{title}",
        headers={"User-Agent": "student-agent-demo/1.0"},
        timeout=10,
    )
    if r.status_code != 200:
        return f"No Wikipedia page found for '{topic}'. Try a simpler or more general title."
    return r.json().get("extract", "No summary available.")

4. Human-in-the-loop agent: nothing is sent until you approve

human_in_loop_agent/agent.py · human_in_loop_agent/tools.py
What it does

Drafts an email, pauses to show it to you, and sends it only after you type approve. Any other answer is treated as feedback, and it redrafts.

How to run
cd day_19\human_in_loop_agent python agent.py
The logic, step by step
  1. draft_node writes the email using the real names. On later rounds it also gets the previous draft and your feedback.
  2. human_review_node calls interrupt({...}), which pauses the graph and hands the draft to the program.
  3. compile(checkpointer=MemorySaver()) is required: it saves the paused state under a thread_id.
  4. run_once() sees __interrupt__ in the result, prints the draft and asks you. Then agent.invoke(Command(resume=answer), config) continues from exactly where it paused.
  5. route_after_review: approve/yes → send, anything else → back to draft.
  6. The send_email tool doesn't really email anyone. It appends the email to outbox.txt.
  7. ask_until_valid() keeps asking until the input makes sense (a valid email address, and an empty Enter never sends).
Flow
START → draft → human_review ⏸ ── approve ──▶ send → END
          ▲            │ your feedback
          └────────────┘
Key idea: Put a human approval step before any risky action, like sending, paying or deleting.

from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langgraph.checkpoint.memory import MemorySaver
from langgraph.types import interrupt, Command
from langchain_core.messages import HumanMessage, SystemMessage
from tools import send_email

OLLAMA_MODEL = "llama3.2"
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.4)

DRAFT_PROMPT = """You are an email assistant. Write a short, polite email (max 80 words)
for the user's request. Start with "Dear <recipient name>," and end with
"Best regards," followed by the sender's name. Use the real names you are given and
NEVER leave placeholders like [Recipient] or [Your Name].
Return ONLY the email body — no subject line, no extra comments."""


# ---------- State ----------
class State(TypedDict):
    request: str
    to: str
    to_name: str
    sender: str
    draft: str
    feedback: str
    result: str


# ---------- Nodes ----------
def draft_node(state: State):
    msg = f"Recipient name: {state['to_name']}\nSender name: {state['sender']}\nRequest: {state['request']}"
    if state.get("feedback"):  # second time around: include the human's notes
        msg += f"\n\nPrevious draft:\n{state['draft']}\n\nHuman feedback — please apply it:\n{state['feedback']}"
    reply = llm.invoke([SystemMessage(content=DRAFT_PROMPT), HumanMessage(content=msg)])
    return {"draft": reply.content.strip()}


def human_review_node(state: State):
    # interrupt() PAUSES the graph here and hands this payload to the human.
    # The graph resumes when we call invoke(Command(resume=...)) with their answer.
    decision = interrupt({"to": state["to"], "draft": state["draft"]})
    return {"feedback": decision}  # "approve" or the human's edit instructions


def route_after_review(state: State):
    return "send" if state["feedback"].strip().lower() in ("approve", "a", "yes", "y") else "draft"


def send_node(state: State):
    result = send_email.invoke({"to": state["to"], "subject": "Message from my assistant", "body": state["draft"]})
    return {"result": result}


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("draft", draft_node)
    g.add_node("human_review", human_review_node)
    g.add_node("send", send_node)

    g.add_edge(START, "draft")
    g.add_edge("draft", "human_review")
    g.add_conditional_edges("human_review", route_after_review, {"send": "send", "draft": "draft"})
    g.add_edge("send", END)
    # A checkpointer is REQUIRED: it saves the paused state so we can resume later.
    return g.compile(checkpointer=MemorySaver())


# ---------- Console helpers (keep input friendly for students) ----------
def ask_until_valid(prompt, is_ok, hint):
    """Keep asking until the answer passes is_ok, so bad input never derails the demo."""
    while True:
        value = input(prompt).strip()
        if is_ok(value):
            return value
        print(f"   ⚠️  {hint}")


def looks_like_email(text):
    local, _, domain = text.partition("@")
    return bool(local) and "." in domain and " " not in text


def run_once(agent, request, to, to_name, sender, thread_id="1"):
    """Drive one email from request to sent — asking the human whenever the graph pauses."""
    config = {"configurable": {"thread_id": thread_id}}
    out = agent.invoke({"request": request, "to": to, "to_name": to_name, "sender": sender,
                        "draft": "", "feedback": "", "result": ""}, config)

    while "__interrupt__" in out:  # graph is paused, waiting for a human
        payload = out["__interrupt__"][0].value
        print(f"\n📧 DRAFT  (to: {to_name} <{payload['to']}>)")
        print("-" * 40)
        print(payload["draft"])
        print("-" * 40)
        # An empty Enter must NOT send — only an explicit answer counts.
        answer = ask_until_valid(
            "👉 Type  approve  to SEND, or type what to change (e.g. 'make it shorter'): ",
            lambda v: v != "", "Please type 'approve' or a change request.")
        out = agent.invoke(Command(resume=answer), config)  # resume from the pause

    return out["result"]


def main():
    print("=" * 50)
    print("  🙋  Human-in-the-Loop Agent  (Ollama + LangGraph)")
    print("=" * 50)
    print(f"  Model: {OLLAMA_MODEL}  |  Nothing is sent until YOU approve\n")
    agent = build_agent()
    sender = ask_until_valid("1) Your name (used to sign the email), e.g. Vamsi: ",
                             lambda v: v != "" and v.lower() not in ("approve", "quit", "q"),
                             "Please type your real name.")
    n = 0
    while True:
        try:
            print("\n--- New email (type q to quit) ---")
            request = ask_until_valid("2) What should the email say? e.g. 'Ask Priya to move tomorrow's meeting to 3pm': ",

        print("\n✅", run_once(agent, request, to, to_name, sender, thread_id=str(n)))

from datetime import datetime
from pathlib import Path

from langchain_core.tools import tool

OUTBOX = Path(__file__).parent / "outbox.txt"


@tool
def send_email(to: str, subject: str, body: str) -> str:
    """
    'Send' an email. For this demo it appends the email to outbox.txt
    (a real app would call SMTP / Gmail API here). This is the RISKY action
    that a human must approve first.
    """
    with OUTBOX.open("a", encoding="utf-8") as f:
        f.write(f"[{datetime.now():%Y-%m-%d %H:%M}] To: {to}\nSubject: {subject}\n{body}\n{'-' * 40}\n")
    return f"Email to {to} saved in {OUTBOX.name}"
