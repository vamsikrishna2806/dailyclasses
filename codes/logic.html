1. three.py: is the number prime?

three.py
What it does

Asks for an integer and says whether it is a prime number.

How to run
python codes/three.py
The logic, step by step
  1. is_prime(number): numbers below 2 are not prime.
  2. It tries every divisor from 2 while divisor * divisor <= number. A larger factor would need a smaller partner, which has already been checked.
  3. If any divisor divides evenly (number % divisor == 0), the number is not prime. Otherwise it is.
Watch out: The last lines are missing an else:, so a prime number prints BOTH "is a prime" and "is not a prime". Put else: before the second print to fix it.
Show the full source of three.py (18 lines)
def is_prime(number):
	if number < 2:
		return False

	divisor = 2
	while divisor * divisor <= number:
		if number % divisor == 0:
			return False
		divisor += 1

	return True


number = int(input("Enter an integer: "))

if is_prime(number):
	print(f"{number} is a prime number.")
	print(f"{number} is not a prime number.")

# Attention Mechanism Simulation
# Shows how AI pays attention to important words

# Simulate which words are "important" to each word
attention_scores = {
    "The": ["bank", "vault", "door"],  # "The" pays attention to these
    "bank": ["vault", "door"],          # "bank" relates to these
    "vault": ["door", "heavy"],         # "vault" relates to these
    "door": ["bank", "vault", "open"],  # "door" relates to these
    "was": ["heavy", "door"],           # "was" describes these
    "heavy": ["door", "open"],          # "heavy" affects these
    "I": ["open", "door"],              # "I" acts on these
    "couldn't": ["open"],               # "couldn't" affects this
    "open": ["door", "it"],             # "open" relates to these
    "it": ["door", "vault"]             # "it" refers to these
}

sentence = "The bank vault door was heavy. I couldn't open it."

print(f"\n📖 Sentence: {sentence}")
print("\n" + "=" * 60)
print("🎯 ATTENTION MECHANISM AT WORK:")
print("=" * 60 + "\n")

for word, attends_to in attention_scores.items():
    print(f"📌 Word: '{word}'")
    print(f"   ↳ Pays attention to: {', '.join(attends_to)}")

    # Show the connection
    if word == "it":
        print(f"   ✨ INSIGHT: '{word}' refers to 'door' (from 'vault door')")
    elif word == "bank":
        print(f"   ✨ INSIGHT: 'bank' = financial (not river) due to 'vault door'")
    print()
