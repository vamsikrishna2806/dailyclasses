Day 20: Ten multi-agent architectures

The same building blocks as Day 19 (State, nodes, edges, conditional edges), arranged into 10 different team shapes. In every folder, agent.py builds the graph and tools.py holds the plain-Python helpers and tools.

Programs on this page
  1. Pipeline (assembly line)
  2. Parallel (fan-out / fan-in)
  3. Debate
  4. Voting panel (majority vote)
  5. Plan and execute
  6. Coder ↔ tester (with a real test run)
  7. RAG team (retrieve, answer, fact-check)
  8. Swarm (hand-offs, no boss)
  9. Blackboard (shared workspace)
  10. Hierarchical (manager → teams → workers)

What every agent.py has in common

1) class State(TypedDict) holds the shared data. 2) Each *_node(state) function makes one LLM call, or runs plain code, and returns the keys it changes. 3) build_agent() adds the nodes and edges, then calls compile(). 4) MAX_* limits guarantee every loop ends. 5) main() asks for input and prints each step.

1. Pipeline (assembly line)

What it does

How to run

  1. The outliner plans the structure. The writer turns the outline into a first draft, the editor polishes it, and the headline node writes a title.

"""
PIPELINE (assembly line) multi-agent
------------------------------------
Four specialist agents work in a FIXED order. Each one reads what the
previous one produced and adds its own piece:

    START -> outliner -> writer -> editor -> headline -> quality_gate -> END
"""
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage
from tools import clean_text, count_words, reading_time

OLLAMA_MODEL = "llama3.2"
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.5)

# ---------- Prompts: one per agent (each agent = one LLM + one job) ----------
OUTLINER_PROMPT = """You are an OUTLINER. Make a 3-point outline for a short blog post about the topic.
One short line per point. Return ONLY the 3 numbered lines."""

WRITER_PROMPT = """You are a BLOG WRITER. Write a short blog post (about 120 words) that follows the outline exactly.
Return ONLY the post text — no title, no intro."""

EDITOR_PROMPT = """You are an EDITOR. Improve the clarity and grammar of the post, remove repetition,
and keep about the same length. Return ONLY the edited post."""

HEADLINE_PROMPT = """You are a HEADLINE WRITER. Write ONE catchy headline (max 10 words) for this post.
Return ONLY the headline — no quotes, no intro."""


# ---------- State: the "baton" passed along the line ----------
class State(TypedDict):
    topic: str
    outline: str
    draft: str
    edited: str
    headline: str
    report: str


def ask_llm(system: str, user: str) -> str:
    return clean_text(llm.invoke([SystemMessage(content=system), HumanMessage(content=user)]).content)


# ---------- Agents (nodes) ----------
def outliner_node(state: State):
    print("  1/5 🗂️  [outliner]  planning the structure...")
    return {"outline": ask_llm(OUTLINER_PROMPT, f"Topic: {state['topic']}")}


def writer_node(state: State):
    print("  2/5 ✍️  [writer]    writing the first draft...")
    msg = f"Topic: {state['topic']}\n\nOutline:\n{state['outline']}"
    return {"draft": ask_llm(WRITER_PROMPT, msg)}


def editor_node(state: State):
    print("  3/5 🧹 [editor]    polishing the draft...")
    return {"edited": ask_llm(EDITOR_PROMPT, state["draft"])}


def headline_node(state: State):
    print("  4/5 📰 [headline]  choosing a headline...")
    return {"headline": ask_llm(HEADLINE_PROMPT, state["edited"]).strip('"')}


def quality_gate_node(state: State):
    """Plain Python (NO LLM): not every node needs AI. This one just measures."""
    print("  5/5 📏 [quality]   measuring the result (plain code, no AI)")
    words = count_words(state["edited"])
    return {"report": f"{words} words · reading time {reading_time(state['edited'])}"}


# ---------- Graph: a straight line ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("outliner", outliner_node)
    g.add_node("writer", writer_node)
    g.add_node("editor", editor_node)
    g.add_node("headline", headline_node)
    g.add_node("quality_gate", quality_gate_node)

    g.add_edge(START, "outliner")
    g.add_edge("outliner", "writer")
    g.add_edge("writer", "editor")
    g.add_edge("editor", "headline")
    g.add_edge("headline", "quality_gate")
    g.add_edge("quality_gate", END)
    return g.compile()


def ask(agent, topic):
    r = agent.invoke({"topic": topic, "outline": "", "draft": "", "edited": "", "headline": "", "report": ""})
    return f"# {r['headline']}\n\n{r['edited']}\n\n({r['report']})"


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print("=" * 52)
    print("  🏭  Pipeline Agent  (outline → write → edit → headline)")
    print("=" * 52)
    print(f"  Model: {OLLAMA_MODEL}")
    print("  Try: 'Why sleep matters' or 'Learning to cook'\n")
    agent = build_agent()
    while True:
        try:
            topic = input("Blog topic: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if not topic: continue
        if topic.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print()
        print("\n" + ask(agent, topic) + "\n")


if __name__ == "__main__":
    main()

import re


def clean_text(text: str) -> str:
    """Small models love to start with 'Here is...:' — drop that intro line."""
    lines = text.strip().splitlines()
    if lines and lines[0].lower().startswith("here") and lines[0].rstrip().endswith(":"):
        lines = lines[1:]
    return "\n".join(lines).strip()


def count_words(text: str) -> int:
    """Count the words in a text."""
    return len(re.findall(r"\b\w[\w'-]*\b", text))


def reading_time(text: str, wpm: int = 200) -> str:
    """Estimate reading time (average adult reads ~200 words per minute)."""
    seconds = round(count_words(text) / wpm * 60)
    return f"{seconds} sec" if seconds < 90 else f"{seconds / 60:.1f} min"

"""
PARALLEL (fan-out / fan-in) multi-agent
---------------------------------------
Three independent specialists work AT THE SAME TIME (fan-out).
When ALL of them finish, one planner merges their results (fan-in).

                 ┌─► flights    ─┐
    START ───────┼─► hotels     ─┼──► planner ─► END
                 └─► activities ─┘
"""
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import create_react_agent
from langchain_core.messages import HumanMessage, SystemMessage
from tools import search_flights, search_hotels, search_activities, find_city

OLLAMA_MODEL = "llama3.2"
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

FLIGHTS_PROMPT = "You are a flight specialist. Call search_flights for the city, then list the options as 2 short bullets."
HOTELS_PROMPT = "You are a hotel specialist. Call search_hotels for the city, then list the options as 2 short bullets."
ACTIVITIES_PROMPT = "You are an activities specialist. Call search_activities for the city, then list the options as short bullets."
PLANNER_PROMPT = """You are a trip planner. You get flight, hotel and activity research for a city.
Write a friendly 3-day itinerary (Day 1, Day 2, Day 3, one line each) and finish with a line
'Budget tip:' naming the cheapest flight and hotel. Use ONLY the facts you were given."""

# Each worker is a small ReAct agent with exactly ONE tool.
flights_agent = create_react_agent(llm, [search_flights])
hotels_agent = create_react_agent(llm, [search_hotels])
activities_agent = create_react_agent(llm, [search_activities])


# ---------- State ----------
# IMPORTANT: parallel nodes must write to DIFFERENT keys, otherwise they overwrite each other.
class State(TypedDict):
    city: str
    flights: str
    hotels: str
    activities: str
    plan: str


def run_worker(agent, prompt, city):
    out = agent.invoke({"messages": [SystemMessage(content=prompt), HumanMessage(content=f"City: {city}")]})
    return out["messages"][-1].content


# ---------- Workers (they run in parallel) ----------
def flights_node(state: State):
    print("  ✈️  [flights]     searching...")
    return {"flights": run_worker(flights_agent, FLIGHTS_PROMPT, state["city"])}


def hotels_node(state: State):
    print("  🏨 [hotels]      searching...")
    return {"hotels": run_worker(hotels_agent, HOTELS_PROMPT, state["city"])}


def activities_node(state: State):
    print("  🎡 [activities]  searching...")
    return {"activities": run_worker(activities_agent, ACTIVITIES_PROMPT, state["city"])}


# ---------- Fan-in: waits until all three are done ----------
def planner_node(state: State):
    print("  🧭 [planner]     all 3 reports arrived — merging into one plan")
    msg = (f"City: {state['city']}\n\nFLIGHTS:\n{state['flights']}\n\n"
           f"HOTELS:\n{state['hotels']}\n\nACTIVITIES:\n{state['activities']}")
    return {"plan": llm.invoke([SystemMessage(content=PLANNER_PROMPT), HumanMessage(content=msg)]).content}


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("flights", flights_node)
    g.add_node("hotels", hotels_node)
    g.add_node("activities", activities_node)
    g.add_node("planner", planner_node)

    # FAN-OUT: three edges leave START → all three nodes start together
    for worker in ("flights", "hotels", "activities"):
        g.add_edge(START, worker)
    # FAN-IN: ONE edge with a LIST of sources → planner waits for ALL of them
    g.add_edge(["flights", "hotels", "activities"], "planner")
    g.add_edge("planner", END)
    return g.compile()


def ask(agent, text):
    city = find_city(text)
    if not city:
        return "I only know: Paris, Tokyo, Goa, New York. Try one of those!"
    return agent.invoke({"city": city, "flights": "", "hotels": "", "activities": "", "plan": ""})["plan"]


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print("=" * 52)
    print("  ⚡  Parallel Agent  (3 specialists at once → 1 planner)")
    print("=" * 52)
    print(f"  Model: {OLLAMA_MODEL}  |  Cities: Paris, Tokyo, Goa, New York\n")
    agent = build_agent()
    while True:
        try:
            text = input("Which city? ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if not text: continue
        if text.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print()
        print("\nTrip plan:\n" + ask(agent, text) + "\n")


if __name__ == "__main__":
    main()

What it does

PRO and CON argue a motion over several rounds, then a neutral judge picks a winner.

How to run
cd day_20_langgraph\debate_agent python agent.py
The logic, step by step
  1. speak() sends the motion, the transcript so far, and a reminder of which side the speaker is on. Small models forget their side, so it is repeated every turn.
  2. Each speech is asked to be at most 50 words, and trim_words cuts it at 60 to be safe. The CON node adds 1 to round, and more_rounds decides whether to loop.
  3. The judge reads the full transcript, and parse_winner pulls out PRO or CON.

                     └─no─▶ judge → END

"""
DEBATE multi-agent
------------------
Two agents with OPPOSITE jobs argue in turns. A neutral judge reads the
whole transcript and picks a winner. Arguing makes both sides stronger.

    START -> pro -> con -> (more rounds?) --yes--> pro ...
                              |
                              no
                              v
                            judge -> END
"""
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage
from tools import trim_words, format_transcript, parse_winner

OLLAMA_MODEL = "llama3.2"
ROUNDS = 2          # how many times each side speaks
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.7)

PRO_PROMPT = """You are the PRO debater. You ARGUE FOR the motion — never switch sides.
Speak in max 50 words. If the opponent already spoke, answer their last point first.
Return ONLY your speech."""

CON_PROMPT = """You are the CON debater. You ARGUE AGAINST the motion — never switch sides.
Speak in max 50 words. Answer the opponent's last point first.
Return ONLY your speech."""

JUDGE_PROMPT = """You are a neutral JUDGE. Read the debate. Decide who argued better
(clear reasons, answered the opponent). Reply in EXACTLY this format:
Winner: PRO   (or)   Winner: CON
Reason: <one short sentence>"""


# ---------- State ----------
class State(TypedDict):
    motion: str
    speeches: list    # every speech so far, e.g. "PRO: ..."
    round: int
    verdict: str
    winner: str


def speak(system: str, state: State, label: str) -> str:
    # Small models forget which side they are on, so we REMIND them every turn.
    stance = "AGREE with" if label == "PRO" else "DISAGREE with"
    msg = (f"Motion: \"{state['motion']}\"\nYour side: {label}. You must {stance} this motion.\n\n"
           f"Debate so far:\n{format_transcript(state['speeches'])}\n\n"
           f"Your turn. Say 'my opponent' instead of PRO/CON. Max 50 words:")
    speech = llm.invoke([SystemMessage(content=system), HumanMessage(content=msg)]).content.strip().strip('"')
    return f"{label}: " + trim_words(speech, 60)


# ---------- Agents ----------
def pro_node(state: State):
    speech = speak(PRO_PROMPT, state, "PRO")
    print(f"\n  🟢 {speech}")
    return {"speeches": state["speeches"] + [speech]}


def con_node(state: State):
    speech = speak(CON_PROMPT, state, "CON")
    print(f"\n  🔴 {speech}")
    return {"speeches": state["speeches"] + [speech], "round": state["round"] + 1}


def judge_node(state: State):
    print("\n  ⚖️  [judge] reading the whole debate...")
    msg = f"Motion: {state['motion']}\n\nDebate:\n{format_transcript(state['speeches'])}"
    verdict = llm.invoke([SystemMessage(content=JUDGE_PROMPT), HumanMessage(content=msg)]).content
    return {"verdict": verdict.strip(), "winner": parse_winner(verdict)}


def more_rounds(state: State) -> str:
    """Conditional edge: keep debating, or call the judge?"""
    return "pro" if state["round"] < ROUNDS else "judge"


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("pro", pro_node)
    g.add_node("con", con_node)
    g.add_node("judge", judge_node)

    g.add_edge(START, "pro")
    g.add_edge("pro", "con")
    g.add_conditional_edges("con", more_rounds, {"pro": "pro", "judge": "judge"})  # the loop
    g.add_edge("judge", END)
    return g.compile()


def ask(agent, motion):
    r = agent.invoke({"motion": motion, "speeches": [], "round": 0, "verdict": "", "winner": ""})
    return r["verdict"]


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print("=" * 52)
    print("  🥊  Debate Agent  (PRO vs CON, then a judge)")
    print("=" * 52)
    print(f"  Model: {OLLAMA_MODEL}  |  Rounds: {ROUNDS}")
    print("  Try: 'Homework should be banned' or 'Cats are better than dogs'\n")
    agent = build_agent()
    while True:
        try:
            motion = input("Motion: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nBye!"); break
        if not motion: continue
        if motion.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print("\n  --- the debate begins ---")
        print("\n⚖️  Verdict:\n" + ask(agent, motion) + "\n")


if __name__ == "__main__":
    main()

def trim_words(text: str, limit: int = 60) -> str:
    """Keep a debater's speech short: cut after `limit` words."""
    words = text.strip().split()
    return " ".join(words[:limit]) + ("…" if len(words) > limit else "")


def format_transcript(speeches: list) -> str:
    """Turn the list of speeches into a readable transcript for the next speaker / judge."""
    return "\n".join(speeches) if speeches else "(no speeches yet)"


def parse_winner(verdict: str) -> str:
    """Find PRO or CON in the judge's reply. Returns 'PRO', 'CON' or 'TIE'."""
    text = verdict.upper()
    pro, con = text.find("WINNER: PRO"), text.find("WINNER: CON")
    if pro == -1 and con == -1:
        return "TIE"
    if pro == -1: return "CON"
    if con == -1: return "PRO"
    return "PRO" if pro < con else "CON"

4. Voting panel (majority vote)

voting_agent/agent.py · voting_agent/tools.py
What it does

Three jurors with different personalities vote on whether a message is spam. A judge is called only if they can't agree.

How to run
cd day_20_langgraph\voting_agent python agent.py
The logic, step by step
  1. The 3 jurors run in parallel. The evidence juror also gets hard facts from spam_signals(): the number of links, urgent words, money words and exclamation marks, the share of capital letters, and a total score.
  2. parse_vote turns each reply into SPAM, NOT_SPAM or UNSURE.
  3. tally() (plain code) needs at least 2 votes for one answer and no tie. Otherwise the judge decides.
  4. One AI can be wrong. A panel is wrong less often.

START ─┬─▶ strict ───┐
        ├─▶ friendly ─┼─▶ tally ─ majority ─▶ END
        └─▶ evidence ─┘         └ none ─▶ judge → END

"""
VOTING PANEL multi-agent  (ensemble / majority vote)
----------------------------------------------------
Ask three jurors with DIFFERENT personalities the SAME question, at the same time.
Count the votes in plain Python. Only if the panel cannot agree, a judge breaks the tie.
One AI can be wrong; a panel is wrong less often.

                ┌─► strict   ─┐
    START ──────┼─► friendly ─┼──► tally ──┬─ clear majority ─► END
                └─► evidence ─┘            └─ no majority ────► judge ─► END
"""
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage
from tools import spam_signals, parse_vote, parse_reply, tally

OLLAMA_MODEL = "llama3.2"
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

# Reason FIRST, vote second: small models vote more consistently after they have said why.
VOTE_FORMAT = "\nReply in exactly two lines:\nReason: <under 12 words>\nVote: SPAM or NOT_SPAM or UNSURE"

STRICT_PROMPT = ("You are a STRICT security guard. You distrust everything: if a message pushes you to click, pay, "
                 "reply with details or hurry, vote SPAM. Decide if the message is spam." + VOTE_FORMAT)
FRIENDLY_PROMPT = ("You are a FRIENDLY, trusting person. You give people the benefit of the doubt: normal chat, "
                   "reminders, receipts and newsletters from known senders are NOT_SPAM. But obvious scams "
                   "(prizes, free gift cards, 'claim now', strange links) are still SPAM. Decide if the message is spam." + VOTE_FORMAT)
EVIDENCE_PROMPT = ("You are a DATA ANALYST. You get a message AND counted warning signs. Follow this rule strictly: "
                   "score 4 or more -> SPAM; score 0 or 1 -> NOT_SPAM; score 2 or 3 -> UNSURE. "
                   "Quote the score in your reason." + VOTE_FORMAT)
JUDGE_PROMPT = """You are the TIE-BREAKER JUDGE. The panel disagreed. Read the message and their reasons and give
the final call. Reply in exactly two lines:
Reason: <one short sentence>
Vote: SPAM or NOT_SPAM"""


# ---------- State: each juror writes to its OWN key (they run in parallel) ----------
class State(TypedDict):
    message: str
    strict: str
    friendly: str
    evidence: str
    counts: str
    verdict: str
    decided_by: str


def juror(system: str, user: str) -> str:
    reply = llm.invoke([SystemMessage(content=system), HumanMessage(content=user)]).content
    return parse_reply(reply)          # -> "SPAM - short reason"


# ---------- Jurors (run at the same time) ----------
def strict_node(state: State):
    return {"strict": juror(STRICT_PROMPT, f"Message: {state['message']}")}


def friendly_node(state: State):
    return {"friendly": juror(FRIENDLY_PROMPT, f"Message: {state['message']}")}


def evidence_node(state: State):
    signals = spam_signals(state["message"])           # a TOOL gives this juror hard facts
    return {"evidence": juror(EVIDENCE_PROMPT, f"Message: {state['message']}\n\nWarning signs: {signals}")}


# ---------- Counting (plain Python) ----------
def tally_node(state: State):
    votes = [parse_vote(state[k]) for k in ("strict", "friendly", "evidence")]
    winner, counts = tally(votes)
    print(f"  🗳️  [strict]   {state['strict']}")

URGENT = ("urgent", "act now", "limited time", "last chance", "immediately", "expires", "hurry")

def spam_signals(text: str) -> dict:
    """Count simple, objective spam warning signs. (Plain code — no AI.)"""
    t = text.lower()
    letters = [c for c in text if c.isalpha()]
    caps = round(sum(c.isupper() for c in letters) / len(letters), 2) if letters else 0
    signals = {
        "links": len(re.findall(r"https?://|www\.|bit\.ly", t)),
        "urgent_words": sum(w in t for w in URGENT),
        "money_words": sum(w in t for w in MONEY),
        "exclamations": text.count("!"),
        "caps_ratio": caps,
    }
    signals["score"] = (signals["links"] * 2 + signals["urgent_words"] * 2 + signals["money_words"] * 2
                        + min(signals["exclamations"], 3) + (2 if caps > 0.4 else 0))
    return signals


def parse_vote(text: str) -> str:
    """Read SPAM / NOT_SPAM / UNSURE from a juror's reply."""
    t = text.upper().replace("NOT SPAM", "NOT_SPAM").replace("NOT-SPAM", "NOT_SPAM")
    first = t.strip()[:30]
    if "NOT_SPAM" in first: return "NOT_SPAM"
    if "SPAM" in first: return "SPAM"
    if "UNSURE" in first: return "UNSURE"
    return "NOT_SPAM" if "NOT_SPAM" in t else "SPAM" if "SPAM" in t else "UNSURE"


def tally(votes: list):
    """Count votes. Returns (winner or None, Counter). A winner needs at least 2 votes."""
    counts = Counter(v for v in votes if v != "UNSURE")
    if counts:
        top, n = counts.most_common(1)[0]
        if n >= 2 and list(counts.values()).count(n) == 1:
            return top, counts
    return None, counts


def parse_reply(text: str) -> str:
    """Turn 'Reason: ... / Vote: SPAM' (two lines) into one tidy line 'SPAM - ...' (vote first, easy to read)."""
    m = re.search(r"vote:\s*([A-Za-z_ -]+)", text, re.I)
    vote = parse_vote(m.group(1)) if m else parse_vote(text)
    r = re.search(r"reason:\s*(.+)", text, re.I)
    reason = (r.group(1) if r else text.strip().splitlines()[0]).strip()
    return f"{vote} - {reason}"

import ast
import operator
import re

from langchain_core.tools import tool

_OPS = {ast.Add: operator.add, ast.Sub: operator.sub, ast.Mult: operator.mul,
        ast.Div: operator.truediv, ast.Pow: operator.pow, ast.USub: operator.neg, ast.Mod: operator.mod}


def _eval(node):
    """Safely evaluate a math expression tree (numbers and + - * / ** % only)."""
    if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
        return node.value
    if isinstance(node, ast.BinOp) and type(node.op) in _OPS:
        return _OPS[type(node.op)](_eval(node.left), _eval(node.right))
    if isinstance(node, ast.UnaryOp) and type(node.op) in _OPS:
        return _OPS[type(node.op)](_eval(node.operand))
    raise ValueError("only numbers and + - * / ** % are allowed")


@tool
def calculator(expression: str) -> str:
    """Calculate a math expression such as '3 * 2 + 4 * 5' or '(26 * 0.9)'. Numbers and + - * / only."""
    try:
        expr = expression.replace("$", "").replace(",", "").replace("%", "/100").strip()
        result = _eval(ast.parse(expr, mode="eval").body)
        return str(round(result, 4))
    except Exception as e:
        return f"Calculator error: {e}. Send a plain expression like 3*2+4*5."


UNITS = {("km", "miles"): 0.621371, ("miles", "km"): 1.60934, ("kg", "lb"): 2.20462,
         ("lb", "kg"): 0.453592, ("m", "ft"): 3.28084, ("ft", "m"): 0.3048}


@tool
def unit_convert(value: float, from_unit: str, to_unit: str) -> str:
    """Convert a value between km/miles, kg/lb or m/ft. Example: value=10, from_unit='km', to_unit='miles'."""
    factor = UNITS.get((from_unit.lower().strip(), to_unit.lower().strip()))
    if factor is None:
        return "Unsupported conversion. Supported: km<->miles, kg<->lb, m<->ft."
    return f"{round(float(value) * factor, 3)} {to_unit}"


def parse_plan(text: str, max_steps: int = 4) -> list:
    """Turn a numbered list ('1. do this', '2. do that') into ['do this', 'do that']."""
    steps = [re.sub(r"^\s*\d+[.)]\s*", "", ln).strip() for ln in text.splitlines() if re.match(r"^\s*\d+[.)]", ln)]
    # Guard rail: a small model sometimes writes a RESULT as a step ("4. 108", "$40 - 30 = $10"). Drop those.
    steps = [s for s in steps if re.search(r"[A-Za-z]{3,}", s) and "=" not in s]
    return steps[:max_steps] or [text.strip()]

import os
import re
import subprocess
import sys
import tempfile


def strip_code_fences(text: str) -> str:
    """Models wrap code in ```python ... ``` — keep only the code inside."""
    m = re.search(r"```(?:python|py)?\s*\n(.*?)```", text, re.S)
    return (m.group(1) if m else text).strip()


def code_blocks(text: str) -> list:
    """Return EVERY ```python ... ``` block found in the text (in order)."""
    return [b.strip() for b in re.findall(r"```(?:python|py)?\s*\n(.*?)```", text, re.S)]


HARNESS = '''
import sys
_failures = []
for _t in __TESTS__.splitlines():
    _t = _t.strip()
    if not _t.startswith("assert"):
        continue
    try:
        exec(_t)
    except AssertionError:
        _msg = _t
        if " == " in _t:                       # show what the function REALLY returned
            _left = _t[len("assert"):].split(" == ", 1)[0]
            try:
                _msg += "    -> but it returned " + repr(eval(_left))
            except Exception as _e:
                _msg += "    -> error: " + str(_e)
        _failures.append(_msg)
    except Exception as _e:
        _failures.append(_t + "    -> " + type(_e).__name__ + ": " + str(_e))
if _failures:
    print("FAILED:")
    print("\\n".join(_failures))
    sys.exit(1)
print("ALL TESTS PASSED")
'''


def build_test_program(code: str, tests: str) -> str:
    """Code + a small harness that runs each assert and reports WHAT went wrong (like a mini pytest)."""
    return code + "\n\n" + HARNESS.replace("__TESTS__", repr(tests))


def run_python(code: str, timeout: int = 5) -> dict:
    """
    Run Python code in a SEPARATE process (with a time limit) and report what happened.
    Returns {"ok": bool, "output": str}.
    ⚠️ Only run code you are comfortable running — this is a classroom demo, not a sandbox.
    """
    with tempfile.TemporaryDirectory() as folder:
        path = os.path.join(folder, "snippet.py")
        with open(path, "w", encoding="utf-8") as f:
            f.write(code)
        try:
            p = subprocess.run([sys.executable, path], capture_output=True, text=True,
                               timeout=timeout, cwd=folder)
        except subprocess.TimeoutExpired:
            return {"ok": False, "output": f"Timed out after {timeout}s (infinite loop?)"}
    output = (p.stdout + p.stderr).strip()
    # keep only the last lines of a traceback — that is where the useful error is
    output = "\n".join(output.splitlines()[-12:])
    return {"ok": p.returncode == 0, "output": output}

  retriever  finds handbook passages (plain code, no AI)
  answerer   answers ONLY from those passages and cites them like [HR-1]
  checker    is every claim really in the passages? if not, send it back

    START -> rewriter -> retriever -> found anything? --no--> no_answer -> END
                                          |yes
                                          v
                            answerer -> checker -> grounded? --yes--> END
                                ^                      |
                                └──── (feedback) ──────┘ no, and rounds left
"""

from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage

OLLAMA_MODEL = "llama3.2"
MAX_ROUNDS = 2        # how many times the checker may send the answer back
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

REWRITER_PROMPT = """You are a SEARCH QUERY WRITER. Turn the user's question into 3-5 keywords for a document search.
Return ONLY the keywords separated by spaces."""

ANSWERER_PROMPT = """You are a HANDBOOK ASSISTANT. Answer the question using ONLY the CONTEXT passages.
Be short (max 2 sentences) and end with the source ids in brackets, like [HR-1].
If the context does not contain the answer, reply exactly: I could not find that in the handbook."""

CHECKER_PROMPT = """You are a FACT CHECKER. Check ONLY the statements made in the ANSWER (ignore other passages).
Look for every fact and number of the ANSWER in the CONTEXT.
- If every fact in the answer appears in the CONTEXT, reply with the single word: GROUNDED
- If the answer says something the CONTEXT does not say, reply: UNSUPPORTED: <the exact claim that is missing>"""


# ---------- State ----------
class State(TypedDict):
    question: str
    search_query: str
    docs: list
    answer: str
    verdict: str
    rounds: int


def say(system: str, user: str) -> str:
    return llm.invoke([SystemMessage(content=system), HumanMessage(content=user)]).content.strip()


# ---------- Agents ----------
def rewriter_node(state: State):
    query = say(REWRITER_PROMPT, state["question"])
    print(f"  🔍 [rewriter]   search keywords: {query}")
    return {"search_query": query}


def retriever_node(state: State):
    """Plain code (NO LLM). Search with the rewritten query AND the original words."""
    docs = search_docs(state["search_query"] + " " + " ".join(keywords(state["question"])))
    print(f"  📚 [retriever]  found: {[d['id'] for d in docs] or 'nothing'}")
    return {"docs": docs}


def no_answer_node(state: State):
    return {"answer": "I could not find that in the handbook.", "verdict": "NO SOURCES"}


def answerer_node(state: State):
    msg = f"CONTEXT:\n{format_context(state['docs'])}\n\nQUESTION: {state['question']}"
    if state["verdict"].startswith("UNSUPPORTED"):
        msg += f"\n\nYour previous answer was rejected: {state['verdict']}\nFix it using only the context."
    answer = say(ANSWERER_PROMPT, msg)
    print(f"  💬 [answerer]   {answer}")
    return {"answer": answer}


def checker_node(state: State):
    if state["answer"].lower().startswith("i could not find"):
        # An honest "I don't know" makes no claims, so there is nothing to fact-check.
        print("  🕵️  [checker]    nothing to check (the answerer admitted it does not know)")
        return {"verdict": "GROUNDED", "rounds": state["rounds"] + 1}
    msg = f"CONTEXT:\n{format_context(state['docs'])}\n\nANSWER: {state['answer']}"
    verdict = say(CHECKER_PROMPT, msg).splitlines()[0]
    if not verdict.upper().startswith("UNSUPPORTED"):
        verdict = "GROUNDED"
    else:
        # Verify the verifier! A small model sometimes flags a "missing" claim that is literally in the passages.
        # Plain Python can settle that: if the flagged text appears in the context, the checker was wrong.
        claim = verdict.split(":", 1)[1].strip(" \"'.") if ":" in verdict else ""
        if len(claim) < 4 or claim.lower() in format_context(state["docs"]).lower():
            verdict = "GROUNDED"
    print(f"  🕵️  [checker]    {verdict}")
    return {"verdict": verdict, "rounds": state["rounds"] + 1}


def after_retrieval(state: State) -> str:
    return "answerer" if state["docs"] else "no_answer"


def after_check(state: State) -> str:
    if state["verdict"] == "GROUNDED" or state["rounds"] > MAX_ROUNDS:
        return "end"
    return "answerer"


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("rewriter", rewriter_node)
    g.add_node("retriever", retriever_node)
    g.add_node("no_answer", no_answer_node)
    g.add_node("answerer", answerer_node)
    g.add_node("checker", checker_node)

    g.add_edge(START, "rewriter")
    g.add_edge("rewriter", "retriever")
    g.add_conditional_edges("retriever", after_retrieval, {"answerer": "answerer", "no_answer": "no_answer"})
    g.add_edge("no_answer", END)
    g.add_edge("answerer", "checker")
    g.add_conditional_edges("checker", after_check, {"end": END, "answerer": "answerer"})   # the loop
    return g.compile()


def ask(agent, question):
    r = agent.invoke({"question": question, "search_query": "", "docs": [], "answer": "", "verdict": "", "rounds": 0})
    badge = {"GROUNDED": "✅ checked against the handbook"}.get(r["verdict"], f"⚠️ {r['verdict']}")
    return f"{r['answer']}\n   ({badge})"


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print("=" * 52)
    print("  📚  RAG Team Agent  (retrieve → answer → fact-check)")

        print()
        print("\nAnswer:", ask(agent, question), "\n")

    main()

STOPWORDS = set("a an the is are do does how many much what when where who can i my me we our of to in on for and or it be at by with you your get".split())


def keywords(text: str) -> list:
    words = re.findall(r"[a-z0-9']+", text.lower())
    # tiny "stemmer": passwords -> password, days -> day (so singular and plural words match)
    words = [w[:-1] if len(w) > 3 and w.endswith("s") else w for w in words]
    return [w for w in words if w not in STOPWORDS and len(w) > 2]


def search_docs(query: str, k: int = 2) -> list:
    """
    Very simple search: score each document by how many query keywords it contains.
    (A real system would use embeddings + a vector store, as you learned on day 12.)
    """
    terms = set(keywords(query))
    scored = []
    for doc in DOCS:
        words = set(keywords(doc["title"] + " " + doc["text"]))
        score = len(terms & words)
        if score:
            scored.append((score, doc))
    scored.sort(key=lambda p: -p[0])
    if not scored:
        return []
    best = scored[0][0]
    # keep only passages that are nearly as relevant as the best one (drops weak matches)
    return [doc for score, doc in scored[:k] if score >= max(1, 0.6 * best)]


def format_context(docs: list) -> str:
    return "\n".join(f"[{d['id']}] {d['title']}: {d['text']}" for d in docs)

"""
SWARM (hand-off) multi-agent
----------------------------
There is NO boss. Every agent decides for itself who should take over next
by ending its reply with `HANDOFF: <agent>` (like passing a phone call on).

    START -> triage ──► orders ──► refunds
               │          │           │
               └──────────┴───────────┴──► DONE -> END
"""
import re
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langchain_core.messages import HumanMessage, SystemMessage
from tools import get_order_status, issue_refund, parse_handoff

OLLAMA_MODEL = "llama3.2"
MAX_HOPS = 5         # safety limit so agents cannot pass the call around forever
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0)

HANDOFF_RULE = """
Your reply has TWO parts: first your message to the customer, then ONE last line that is exactly one of:
HANDOFF: orders
HANDOFF: refunds
HANDOFF: DONE"""

TRIAGE_PROMPT = """You are the FRONT DESK of an online shop. Write ONE short friendly sentence to the customer.
If the message mentions an order, delivery, damage or a refund, do NOT ask questions — pass it on with HANDOFF: orders.
If it is only small talk, answer it and use HANDOFF: DONE.

Example reply:
Sorry about that, let me get our orders team.
HANDOFF: orders""" + HANDOFF_RULE

ORDERS_PROMPT = """You are the ORDERS agent. You get the customer's message and the TOOL RESULT for their order.
Tell the customer the status in 1-2 sentences.
If the customer asks for a refund / their money back AND the status is 'delivered' -> HANDOFF: refunds
(you cannot refund yourself — the refunds agent does it). Otherwise -> HANDOFF: DONE.""" + HANDOFF_RULE

REFUNDS_PROMPT = """You are the REFUNDS agent. You get the customer's message and the TOOL RESULT of the refund.
Tell the customer the outcome in 1-2 sentences, using the tool result. Then HANDOFF: DONE.""" + HANDOFF_RULE


# ---------- State ----------
class State(TypedDict):
    request: str
    active: str      # which agent speaks next
    log: list        # replies so far: ["[triage] ...", "[orders] ..."]
    hops: int


# Small models sometimes forget the HANDOFF line. Safety net: what to do when it is missing.
SAFETY_NET = {"triage": "orders"}          # triage's whole job is to pass the customer on
ICONS = {"triage": "🛎️ ", "orders": "📦", "refunds": "💸"}


def order_id_from(text: str) -> str:
    """Find a 4-digit order number such as 5001 in the customer's message."""
    m = re.search(r"\b(\d{4})\b", text)
    return m.group(1) if m else ""


def conversation(state: State) -> str:
    seen = "\n".join(state["log"]) or "(nobody has replied yet)"
    return f"Customer message: {state['request']}\n\nWhat the team already said:\n{seen}"


def finish_turn(state: State, name: str, raw: str):
    """Every agent ends the same way: record the reply and choose who is next."""
    message, target = parse_handoff(raw, default=SAFETY_NET.get(name, "done"))
    message = message or "(no message)"
    if target == name or state["hops"] + 1 >= MAX_HOPS:      # guard rails
        target = "done"
    print(f"  {ICONS[name]} [{name}] {message}")
    print(f"       ↪ hands off to: {target}")
    return {"log": state["log"] + [f"[{name}] {message}"], "active": target, "hops": state["hops"] + 1}


# ---------- Agents ----------
def triage_node(state: State):
    raw = llm.invoke([SystemMessage(content=TRIAGE_PROMPT), HumanMessage(content=conversation(state))]).content
    return finish_turn(state, "triage", raw)


def orders_node(state: State):
    # Tool-first: plain code finds the order number and runs the tool; the LLM explains the result.
    result = get_order_status.invoke({"order_id": order_id_from(state["request"])})
    print(f"       🔧 tool get_order_status -> {result}")
    msg = conversation(state) + f"\n\nTOOL RESULT: {result}"
    raw = llm.invoke([SystemMessage(content=ORDERS_PROMPT), HumanMessage(content=msg)]).content
    return finish_turn(state, "orders", raw)


def refunds_node(state: State):
    result = issue_refund.invoke({"order_id": order_id_from(state["request"])})
    print(f"       🔧 tool issue_refund -> {result}")
    msg = conversation(state) + f"\n\nTOOL RESULT: {result}"
    raw = llm.invoke([SystemMessage(content=REFUNDS_PROMPT), HumanMessage(content=msg)]).content
    return finish_turn(state, "refunds", raw)


def next_agent(state: State) -> str:
    """Conditional edge: read the agent's own HANDOFF decision."""
    return state["active"] if state["active"] in ("triage", "orders", "refunds") else "done"


# ---------- Graph ----------
def build_agent():
    g = StateGraph(State)
    g.add_node("triage", triage_node)
    g.add_node("orders", orders_node)
    g.add_node("refunds", refunds_node)

    g.add_edge(START, "triage")
    routes = {"triage": "triage", "orders": "orders", "refunds": "refunds", "done": END}
    for name in ("triage", "orders", "refunds"):       # EVERY agent can hand off to any other
        g.add_conditional_edges(name, next_agent, routes)
    return g.compile()


def ask(agent, request):
    r = agent.invoke({"request": request, "active": "triage", "log": [], "hops": 0})
    return "\n".join(r["log"])

REFUND_LOG = []   # every approved refund is recorded here


@tool
def get_order_status(order_id: str) -> str:
    """Look up an order by its number (e.g. 5001) and return item, status and total."""
    o = ORDERS.get(order_id.strip())
    if not o:
        return f"No order found with id '{order_id}'."
    return f"Order {order_id}: {o['item']}, ${o['total']:.2f}, status = {o['status']} ({o['note']})."


@tool
def issue_refund(order_id: str) -> str:
    """Refund an order by its number. Only DELIVERED orders can be refunded."""
    o = ORDERS.get(order_id.strip())
    if not o:
        return f"Refund failed: no order '{order_id}'."
    if o["status"] != "delivered":
        return f"Refund refused: order {order_id} is '{o['status']}', only delivered orders can be refunded."

    REFUND_LOG.append(order_id)
    return f"Refund approved: ${o['total']:.2f} for order {order_id} returns to the customer's card in 3-5 days."


def parse_handoff(text: str, default: str = "done"):
    """
    Split an agent reply into (message for the customer, who is next).
    The agent ends with a line like 'HANDOFF: refunds'. No valid line => `default`.
    """
    matches = re.findall(r"HANDOFF:\s*(orders|refunds|triage|done)", text, re.I)
    target = matches[-1].lower() if matches else default
    message = re.sub(r"HANDOFF:.*", "", text, flags=re.I).strip()
    return message, target

BLACKBOARD multi-agent  (shared workspace)
------------------------------------------
Imagine a whiteboard in a meeting room. Specialists do NOT talk to each other and
nobody tells them what to do. Each specialist only watches the board:
"Is there something here I can add to?"  If yes, it writes its part on the board.

        nutritionist  needs: request           adds: constraints

Return ONLY this format, with no extra comments, no totals, no notes:
Day 1: <dish name>
- ingredient
- ingredient
Day 2: ...
If there is BUDGET FEEDBACK, the last menu was too expensive: use mostly items that cost $2.5 or less."""

class State(TypedDict):

    next: str        # who the controller woke up
    steps: int


def new_board(request: str) -> dict:

            "ingredients": [], "cost": 0.0, "verdict": "", "feedback": "", "revisions": 0}

RULES = {
    "nutritionist": lambda b: not b["constraints"],
    "chef":         lambda b: b["constraints"] and not b["menu"],
    "shopper":      lambda b: b["menu"] and not b["ingredients"],
    "auditor":      lambda b: b["ingredients"] and not b["verdict"],
}


def controller_node(state: State):
    """No AI here. First specialist whose rule is true wakes up. Nobody can? We are done."""
    choice = next((name for name, can in RULES.items() if can(state["board"])), "DONE")
    if state["steps"] >= MAX_STEPS:
        choice = "DONE"
    print(f"  📋 [board] -> {choice}")
    return {"next": choice, "steps": state["steps"] + 1}

def nutritionist_node(state: State):

    b["constraints"] = say(NUTRITIONIST_PROMPT, b["request"])

def chef_node(state: State):
    b = dict(state["board"])
    msg = (f"Request: {b['request']}\nBUDGET FOR THE WHOLE MENU: ${b['budget']:.0f}\n\n"

    if b["feedback"]:
        msg += f"\n\nBUDGET FEEDBACK: {b['feedback']} Make it cheaper!"
    b["menu"] = say(CHEF_PROMPT, msg)
    print("       👨‍🍳 chef wrote a menu on the board" + (" (cheaper version)" if b["feedback"] else ""))

    items = parse_list(say(SHOPPER_PROMPT, b["menu"]))
    b["ingredients"] = items
    b["cost"] = round(sum(price_of(i) for i in items), 2)
    print(f"       🛒 shopper listed {len(items)} items, total ${b['cost']:.2f}")
    return {"board": b}


def auditor_node(state: State):
    """Plain Python judge: compares cost with budget. Too expensive? ERASE the menu so the chef is re-triggered."""
    b = dict(state["board"])
    if b["cost"] > b["budget"] and b["revisions"] < MAX_REVISIONS:
        print(f"       🧾 auditor: ${b['cost']:.2f} is over ${b['budget']:.0f} -> erasing menu, chef must redo it")
        priciest = sorted(b["ingredients"], key=price_of, reverse=True)[:3]       # be SPECIFIC so the chef can act
        pricey_text = ", ".join(f"{i} (${price_of(i):g})" for i in priciest)

    g.add_edge(START, "controller")
    g.add_conditional_edges("controller", lambda s: s["next"],
                            {"nutritionist": "nutritionist", "chef": "chef", "shopper": "shopper",

    """Turn '- chicken\\n- rice' into ['chicken', 'rice'] (no duplicates, original order)."""

"""
HIERARCHICAL multi-agent  (a company: manager -> team leads -> workers)
-----------------------------------------------------------------------
The top MANAGER only talks to TEAMS. Each team is its OWN small graph with its own workers
(a graph inside a graph). Teams report back to the manager.

                       ┌──────── product team ─────────┐
    START -> manager ──┤  researcher -> spec_writer -> lead │──┐
                ^      └────────────────────────────────┘    │
                │      ┌──────── marketing team ───────┐     │
                ├──────┤  copywriter -> social_planner -> lead │
                │      └────────────────────────────────┘    │
                └───────────────────(report back)────────────┘
                manager says FINISH -> END
"""
import sys
from typing import TypedDict

from langchain_ollama import ChatOllama
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import create_react_agent
from langchain_core.messages import HumanMessage, SystemMessage
from tools import audience_insights

OLLAMA_MODEL = "llama3.2"
MAX_STEPS = 5
llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.3)

MANAGER_PROMPT = """You are the MANAGER of a company with two teams:
  product   -> researches the audience and writes a product spec
  marketing -> writes the tagline, pitch and social posts
Decide which team works NEXT. Product comes first, then marketing, then FINISH.
Answer with EXACTLY ONE word: product, marketing, or FINISH."""

RESEARCHER_PROMPT = "You are a RESEARCHER. Call audience_insights with the product idea, then report the audience in 2 short bullets."
SPEC_WRITER_PROMPT = "You are a SPEC WRITER. From the idea and audience facts, write 3 short bullet points: the key features. Return only the bullets."
PRODUCT_LEAD_PROMPT = "You are the PRODUCT LEAD. Merge the research and features into a 'PRODUCT SPEC' of 4 short lines. Return only the spec."
COPYWRITER_PROMPT = "You are a COPYWRITER. Using the product spec, write a tagline (max 6 words) and a 2-sentence pitch. Return only those."
SOCIAL_PLANNER_PROMPT = "You are a SOCIAL PLANNER. Write 3 short social media posts (one line each, with an emoji) for this product."
MARKETING_LEAD_PROMPT = "You are the MARKETING LEAD. Combine the copy and posts into a tidy 'CAMPAIGN' section. Return only the campaign."

researcher_agent = create_react_agent(llm, [audience_insights])


def say(system: str, user: str) -> str:
    return llm.invoke([SystemMessage(content=system), HumanMessage(content=user)]).content.strip()


# =====================================================================
# LEVEL 2 — each TEAM is its own graph with its own small state
# =====================================================================
class TeamState(TypedDict):
    brief: str     # what the team was asked to do (+ context from other teams)
    step1: str     # first worker's output
    step2: str     # second worker's output
    report: str    # the lead's final report


def build_product_team():
    def researcher(s: TeamState):
        print("      🔎 [researcher]     asking the audience tool...")
        out = researcher_agent.invoke({"messages": [SystemMessage(content=RESEARCHER_PROMPT), HumanMessage(content=s["brief"])]})
        return {"step1": out["messages"][-1].content}

    def spec_writer(s: TeamState):
        print("      📝 [spec_writer]    drafting features...")
        return {"step2": say(SPEC_WRITER_PROMPT, f"Idea: {s['brief']}\n\nAudience:\n{s['step1']}")}

    def lead(s: TeamState):
        print("      👩‍💼 [product_lead]  merging into the product spec...")
        return {"report": say(PRODUCT_LEAD_PROMPT, f"Audience:\n{s['step1']}\n\nFeatures:\n{s['step2']}")}

    return _linear_team(researcher, spec_writer, lead)


def build_marketing_team():
    def copywriter(s: TeamState):
        print("      ✍️  [copywriter]     writing tagline + pitch...")
        return {"step1": say(COPYWRITER_PROMPT, s["brief"])}

    def social_planner(s: TeamState):
        print("      📣 [social_planner] writing social posts...")
        return {"step2": say(SOCIAL_PLANNER_PROMPT, s["brief"])}

    def lead(s: TeamState):
        print("      👨‍💼 [marketing_lead] assembling the campaign...")
        return {"report": say(MARKETING_LEAD_PROMPT, f"Copy:\n{s['step1']}\n\nPosts:\n{s['step2']}")}

    return _linear_team(copywriter, social_planner, lead)


def _linear_team(worker1, worker2, lead):
    g = StateGraph(TeamState)
    g.add_node("worker1", worker1)
    g.add_node("worker2", worker2)
    g.add_node("lead", lead)
    g.add_edge(START, "worker1")
    g.add_edge("worker1", "worker2")
    g.add_edge("worker2", "lead")
    g.add_edge("lead", END)
    return g.compile()           # <- a finished graph that we will use as ONE node below


product_team = build_product_team()
marketing_team = build_marketing_team()


# =====================================================================
# LEVEL 1 — the manager's graph only sees two big nodes: the teams
# =====================================================================
class State(TypedDict):
    idea: str
    product: str       # product team's report
    marketing: str     # marketing team's report
    next: str
    steps: int


def manager_node(state: State):
    progress = (f"Idea: {state['idea']}\nProduct team done: {'yes' if state['product'] else 'no'}\n"
                f"Marketing team done: {'yes' if state['marketing'] else 'no'}")
    word = say(MANAGER_PROMPT, progress).lower()
    choice = next((t for t in ("product", "marketing") if t in word), "FINISH")
    # Guard rails: never repeat a team that is already done.
    if choice == "product" and state["product"]:
        choice = "marketing"
    if choice == "marketing" and state["marketing"]:
        choice = "FINISH"
    if not state["product"]:
        choice = "product"                   # marketing needs the spec first
    if state["product"] and state["marketing"] or state["steps"] >= MAX_STEPS:
        choice = "FINISH"
    print(f"  🏢 [manager] -> {choice}")
    return {"next": choice, "steps": state["steps"] + 1}


def product_node(state: State):
    r = product_team.invoke({"brief": state["idea"], "step1": "", "step2": "", "report": ""})
    return {"product": r["report"]}


def marketing_node(state: State):
    brief = f"Product idea: {state['idea']}\n\n{state['product']}"
    r = marketing_team.invoke({"brief": brief, "step1": "", "step2": "", "report": ""})
    return {"marketing": r["report"]}


def build_agent():
    g = StateGraph(State)
    g.add_node("manager", manager_node)
    g.add_node("product", product_node)
    g.add_node("marketing", marketing_node)
    g.add_edge(START, "manager")
    g.add_conditional_edges("manager", lambda s: s["next"],
                            {"product": "product", "marketing": "marketing", "FINISH": END})
    g.add_edge("product", "manager")      # teams report back to the manager
    g.add_edge("marketing", "manager")
    return g.compile()


def ask(agent, idea):
    r = agent.invoke({"idea": idea, "product": "", "marketing": "", "next": "", "steps": 0})
    return f"{r['product']}\n\n{r['marketing']}"


def main():
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    print("=" * 52)
    print("  🏢  Hierarchical Agent  (manager → teams → workers)")
    print("=" * 52)
    print(f"  Model: {OLLAMA_MODEL}  |  Teams: product (3 agents), marketing (3 agents)")
    print("  Try: 'a note-taking app' or 'a language learning app'\n")
    agent = build_agent()
    while True:
        try:
            idea = input("Product idea: ").strip()
        except (EOFError, KeyboardInterrupt):

        if not idea: continue
        if idea.lower() in ("quit", "exit", "q"):
            print("Bye!"); break
        print()

if __name__ == "__main__":
    main()
