"""
=============================================================================
LAYER: Database Persistence Layer (db.py)
DESCRIPTION:
  This file manages all interactions with the SQLite database.
  It uses Python's built-in `sqlite3` module with 100% hand-crafted SQL
  queries (no ORM). It is responsible for database initialization, schema
  creation, seed data insertion, and all CRUD (Create, Read, Update, Delete)
  operations on the `tasks` table.
=============================================================================
"""

import os
import sqlite3
from datetime import datetime

# Resolve database file path relative to this file's directory
DB_PATH = os.path.join(os.path.dirname(os.path.abspath(__file__)), "todo.db")


def get_db_connection():
    """
    Establish and return a new SQLite database connection.
    Configures `row_factory` to `sqlite3.Row` so query results can be
    accessed like dictionaries with column names.
    """
    conn = sqlite3.connect(DB_PATH)
    # Enable dict-like column access for fetched rows
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    """
    Initialize the database table and seed initial sample data if empty.
    Creates the `tasks` table with fields:
      - id: unique auto-incrementing integer identifier
      - title: task description string (cannot be null)
      - done: integer status flag (0 = active, 1 = completed)
      - created_at: ISO-formatted timestamp of task creation
      - due_date: optional ISO date string for task deadline (stretch goal)
    """
    conn = get_db_connection()
    cursor = conn.cursor()

    # Create the tasks table if it does not already exist
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS tasks (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            title TEXT NOT NULL,
            done INTEGER NOT NULL DEFAULT 0,
            created_at TEXT NOT NULL,
            due_date TEXT
        )
    """)

    # Check if the database has any existing tasks
    cursor.execute("SELECT COUNT(*) AS count FROM tasks")
    count = cursor.fetchone()["count"]

    # Seed initial sample data if the table is freshly created and empty
    if count == 0:
        sample_tasks = [
            (
                "Explore the Neumorphic To-Do App design",
                1,
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                datetime.now().strftime("%Y-%m-%d"),
            ),
            (
                "Test adding a new task with due date",
                0,
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                datetime.now().strftime("%Y-%m-%d"),
            ),
            (
                "Filter tasks by Active and Done status",
                0,
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                None,
            ),
            (
                "Review Flask REST API endpoints in app.py",
                0,
                datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                None,
            ),
        ]

        # Insert sample rows using parameterized SQL to prevent SQL injection
        cursor.executemany(
            """
            INSERT INTO tasks (title, done, created_at, due_date)
            VALUES (?, ?, ?, ?)
            """,
            sample_tasks,
        )

        conn.commit()

    conn.close()


def task_to_dict(row):
    """
    Helper function to convert a sqlite3.Row object into a standard Python dict.
    Converts SQLite integer boolean (0 or 1) to a Python boolean for clean JSON serialization.
    """
    if row is None:
        return None
    return {
        "id": row["id"],
        "title": row["title"],
        "done": bool(row["done"]),
        "created_at": row["created_at"],
        "due_date": row["due_date"],
    }


def get_all_tasks():
    """
    Retrieve all tasks from the database ordered by completion status (active first)
    and then descending by ID (newest first).
    Returns a list of task dictionaries.
    """
    conn = get_db_connection()
    cursor = conn.cursor()

    cursor.execute("""
        SELECT id, title, done, created_at, due_date
        FROM tasks
        ORDER BY done ASC, id DESC
    """)
    rows = cursor.fetchall()
    conn.close()

    return [task_to_dict(row) for row in rows]


def get_task_by_id(task_id):
    """
    Fetch a single task record by its unique ID.
    Returns the task dictionary if found, or None if not found.
    """
    conn = get_db_connection()
    cursor = conn.cursor()

    cursor.execute(
        "SELECT id, title, done, created_at, due_date FROM tasks WHERE id = ?",
        (task_id,),
    )
    row = cursor.fetchone()
    conn.close()

    return task_to_dict(row)


def create_task(title, due_date=None):
    """
    Insert a new task into the database.
    - title: text description (validated by caller)
    - due_date: optional string in YYYY-MM-DD format
    Returns the newly inserted task dictionary.
    """
    created_at = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    conn = get_db_connection()
    cursor = conn.cursor()

    cursor.execute(
        """
        INSERT INTO tasks (title, done, created_at, due_date)
        VALUES (?, 0, ?, ?)
        """,
        (title.strip(), created_at, due_date.strip() if due_date else None),
    )
    new_id = cursor.lastrowid
    conn.commit()
    conn.close()

    return get_task_by_id(new_id)


def update_task(task_id, done=None, title=None, due_date=None):
    """
    Update specific attributes of an existing task.
    Supports updating 'done' status, 'title', and/or 'due_date'.
    Returns the updated task dictionary if found, or None if task does not exist.
    """
    # Check if task exists first
    task = get_task_by_id(task_id)
    if not task:
        return None

    # Dynamically build the SQL UPDATE clause based on provided arguments
    update_clauses = []
    params = []

    if done is not None:
        # Convert boolean to integer (1 for True, 0 for False)
        update_clauses.append("done = ?")
        params.append(1 if done else 0)

    if title is not None:
        update_clauses.append("title = ?")
        params.append(title.strip())

    if due_date is not None:
        update_clauses.append("due_date = ?")
        params.append(due_date.strip() if due_date else None)

    # If no fields were provided to update, return the current task
    if not update_clauses:
        return task

    # Append task_id for the WHERE clause
    params.append(task_id)
    sql = f"UPDATE tasks SET {', '.join(update_clauses)} WHERE id = ?"

    conn = get_db_connection()
    cursor = conn.cursor()
    cursor.execute(sql, tuple(params))
    conn.commit()
    conn.close()

    return get_task_by_id(task_id)


def delete_task(task_id):
    """
    Delete a task by its ID.
    Returns True if a record was deleted, or False if no matching record was found.
    """
    conn = get_db_connection()
    cursor = conn.cursor()

    cursor.execute("DELETE FROM tasks WHERE id = ?", (task_id,))
    deleted = cursor.rowcount > 0
    conn.commit()
    conn.close()

    return deleted

@app.route("/api/tasks", methods=["POST"])
def add_task():
    """
    POST /api/tasks
    Create a new task record.
    Expected JSON Body:
      - title: string (required, non-empty)
      - due_date: string (optional, format YYYY-MM-DD)
    Returns:
      - 201 Created: The newly created task object.
      - 400 Bad Request: If title is missing, not a string, or empty.
    """
    # Parse request JSON payload
    data = request.get_json(silent=True)
    if not data or not isinstance(data, dict):
        return jsonify({"error": "Invalid JSON payload provided."}), 400

    title = data.get("title")
    due_date = data.get("due_date")

    # Validate that the title is provided and not just whitespace
    if not title or not isinstance(title, str) or not title.strip():
        return jsonify({"error": "Task title is required and cannot be empty."}), 400

    # Optional due_date validation: must be string or None
    if due_date is not None and not isinstance(due_date, str):
        return jsonify({"error": "Due date must be a valid date string."}), 400

    # Insert into the database
    new_task = create_task(title=title, due_date=due_date)
    return jsonify(new_task), 201


@app.route("/api/tasks/<int:task_id>", methods=["PATCH"])
def patch_task(task_id):
    """
    PATCH /api/tasks/<id>
    Update an existing task (e.g. toggle completion status, update title/due_date).
    Expected JSON Body:
      - done: boolean (optional)
      - title: string (optional)
      - due_date: string or null (optional)
    Returns:
      - 200 OK: The updated task object.
      - 400 Bad Request: If no valid update fields are provided.
      - 404 Not Found: If task with the given ID does not exist.
    """
    # Verify task existence first
    existing_task = get_task_by_id(task_id)
    if not existing_task:
        return jsonify({"error": f"Task with id {task_id} not found."}), 404

    # Parse request JSON payload
    data = request.get_json(silent=True)
    if not data or not isinstance(data, dict):
        return jsonify({"error": "Invalid JSON payload provided."}), 400

    # Extract optional fields to update
    done = data.get("done")
    title = data.get("title")
    due_date = data.get("due_date")

    # Ensure at least one update field was provided
    if done is None and title is None and "due_date" not in data:
        return jsonify({"error": "At least one field ('done', 'title', 'due_date') must be provided."}), 400

    # Validate done if passed
    if done is not None and not isinstance(done, (bool, int)):
        return jsonify({"error": "Field 'done' must be a boolean or integer (0 or 1)."}), 400

    # Validate title if passed
    if title is not None:
        if not isinstance(title, str) or not title.strip():
            return jsonify({"error": "Title cannot be empty."}), 400

    # Update record in SQLite
    updated_task = update_task(task_id, done=done, title=title, due_date=due_date)
    return jsonify(updated_task), 200


@app.route("/api/tasks/<int:task_id>", methods=["DELETE"])
def remove_task(task_id):
    """
    DELETE /api/tasks/<id>
    Remove a task by ID.
    Returns:
      - 200 OK: Confirmation message and deleted task ID.
      - 404 Not Found: If task with given ID does not exist.
    """
    # Verify task existence first
    existing_task = get_task_by_id(task_id)
    if not existing_task:
        return jsonify({"error": f"Task with id {task_id} not found."}), 404

    # Perform deletion
    deleted = delete_task(task_id)
    if not deleted:
        return jsonify({"error": f"Failed to delete task with id {task_id}."}), 500

    return jsonify({"message": "Task deleted successfully.", "id": task_id}), 200

function formatDueDateBadge(dateString, isDone) {
  if (!dateString) return "";

  // Parse target date and today's date (ignoring time component)
  const targetDate = new Date(dateString + "T00:00:00");
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const diffTime = targetDate.getTime() - today.getTime();
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

  let statusClass = "";
  let displayText = "";

  if (diffDays < 0 && !isDone) {
    statusClass = "overdue";
    displayText = `Overdue (${Math.abs(diffDays)}d ago)`;
  } else if (diffDays === 0) {
    statusClass = "today";
    displayText = "Due Today";
  } else if (diffDays === 1) {
    displayText = "Due Tomorrow";
  } else {
    // Format as Month Day (e.g., Oct 25)
    displayText = targetDate.toLocaleDateString(undefined, {
      month: "short",
      day: "numeric",
    });
  }

  return `
    <span class="due-badge ${statusClass}" title="Due: ${dateString}">
      <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
        <rect x="3" y="4" width="18" height="18" rx="2" ry="2"></rect>
        <line x1="16" y1="2" x2="16" y2="6"></line>
        <line x1="8" y1="2" x2="8" y2="6"></line>
        <line x1="3" y1="10" x2="21" y2="10"></line>
      </svg>
      ${displayText}
    </span>
  `;
}

  // Update filter counts
  elements.countAll.textContent = total;
  elements.countActive.textContent = activeCount;
  elements.countDone.textContent = doneCount;
}

      elements.tasksList.appendChild(li);
    });
  }

  // Update counters and progress bar
  updateMetrics();
}

/**
 * Initial load of all tasks from Flask REST API
 */
async function loadTasks() {
  state.isLoading = true;
  elements.tasksLoading.classList.remove("hidden");
  elements.tasksList.classList.add("hidden");

  try {
    const tasks = await API.getAllTasks();
    state.tasks = tasks;
    renderTasks();
  } catch (error) {
    console.error("Error loading tasks:", error);
    showToast(error.message || "Failed to load tasks from server", "danger");
  } finally {
    state.isLoading = false;
    elements.tasksLoading.classList.add("hidden");
    elements.tasksList.classList.remove("hidden");
  }
}

/**
 * Requirement 2: Add a new task with text box and "Add" button
 */
async function handleAddTask(e) {
  e.preventDefault();

  const title = elements.taskTitleInput.value.trim();
  const dueDate = elements.taskDateInput.value;

  // Client validation
  if (!title) {
    elements.formError.textContent = "Please enter a task description.";
    elements.formError.classList.remove("hidden");
    elements.taskTitleInput.focus();
    return;
  }

  elements.formError.classList.add("hidden");
  elements.addTaskBtn.disabled = true;

  try {
    // Call POST /api/tasks
    const createdTask = await API.createTask(title, dueDate);

    // Add to local state (active tasks appear at the top)
    state.tasks.unshift(createdTask);

    // Re-render tasks and update progress
    renderTasks();

    // Reset input fields
    elements.taskTitleInput.value = "";
    elements.taskDateInput.value = "";
    elements.taskTitleInput.focus();

    showToast("Task added successfully!", "success");
  } catch (error) {
    console.error("Error adding task:", error);
    showToast(error.message || "Could not add task", "danger");
  } finally {
    elements.addTaskBtn.disabled = false;
  }
}

/**
 * Requirement 3: Toggle task completion status (strike-through style)
 */
async function handleToggleDone(taskId, newDoneStatus) {
  try {
    // Optimistically update local state for immediate snappy UI feedback
    const taskIndex = state.tasks.findIndex((t) => t.id === taskId);
    if (taskIndex === -1) return;

    const previousDone = state.tasks[taskIndex].done;
    state.tasks[taskIndex].done = newDoneStatus;
    renderTasks();

    // Call PATCH /api/tasks/<id>
    const updated = await API.updateTask(taskId, { done: newDoneStatus });

    // Sync with returned server state
    state.tasks[taskIndex] = updated;
    updateMetrics();

    showToast(
      newDoneStatus ? "Task marked as done!" : "Task marked as active",
      "info"
    );
  } catch (error) {
    console.error("Error updating task status:", error);
    showToast(error.message || "Failed to update task", "danger");
    // Reload tasks on error to guarantee consistency with database
    loadTasks();
  }
}

/**
 * Requirement 4: Delete a task
 */
async function handleDeleteTask(taskId) {
  const taskElement = document.getElementById(`task-item-${taskId}`);

  // Play smooth deletion exit animation
  if (taskElement) {
    taskElement.classList.add("deleting");
  }

  try {
    // Call DELETE /api/tasks/<id>
    await API.deleteTask(taskId);

    // Remove from local state after small delay for animation
    setTimeout(() => {
      state.tasks = state.tasks.filter((t) => t.id !== taskId);
      renderTasks();
      showToast("Task deleted", "danger");
    }, 200);
  } catch (error) {
    console.error("Error deleting task:", error);
    showToast(error.message || "Failed to delete task", "danger");
    if (taskElement) {
      taskElement.classList.remove("deleting");
    }
  }
}

/**
 * Stretch Goal: Filter task list by All / Active / Done
 */
function handleFilterChange(filterValue) {
  state.currentFilter = filterValue;

  // Update active pill UI styling
  elements.filterPills.forEach((pill) => {
    const isSelected = pill.getAttribute("data-filter") === filterValue;
    pill.classList.toggle("active", isSelected);
    pill.setAttribute("aria-selected", isSelected ? "true" : "false");
  });

  renderTasks();
}

// ---------------------------------------------------------------------------
// Setup Event Listeners
// ---------------------------------------------------------------------------
function setupEventListeners() {
  // Add task form submission
  elements.addTaskForm.addEventListener("submit", handleAddTask);

  // Clear form error as user types
  elements.taskTitleInput.addEventListener("input", () => {
    if (!elements.formError.classList.contains("hidden")) {
      elements.formError.classList.add("hidden");
    }
  });

  // Filter tabs click handling
  elements.filterPills.forEach((pill) => {
    pill.addEventListener("click", () => {
      const filter = pill.getAttribute("data-filter");
      handleFilterChange(filter);
    });
  });

  // Delegated event listener for task items (checkboxes and delete buttons)
  elements.tasksList.addEventListener("click", (e) => {
    // Check if delete button was clicked
    const deleteBtn = e.target.closest('[data-action="delete-task"]');
    if (deleteBtn) {
      const taskId = parseInt(deleteBtn.getAttribute("data-id"), 10);
      handleDeleteTask(taskId);
      return;
    }

    // Check if checkbox was toggled
    const checkbox = e.target.closest('[data-action="toggle-done"]');
    if (checkbox) {
      const taskId = parseInt(checkbox.getAttribute("data-id"), 10);
      const isChecked = checkbox.checked;
      handleToggleDone(taskId, isChecked);
    }
  });
}

// Application Bootstrap

document.addEventListener("DOMContentLoaded", () => {
  setupEventListeners();
  loadTasks();
