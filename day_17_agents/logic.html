Every agent folder has agent.py (the loop) and tools.py (the hands). agent.py works like this:

1. agent.py: the common pattern

wikipedia_agent/agent.py
What it does

Builds a ReAct agent from a local Ollama model and one or two tools, then chats in a loop.

The logic, step by step
  1. ChatOllama(model="llama3.2", temperature=0.3) is the brain.
  2. create_react_agent(llm, [tool]) (LangGraph's prebuilt agent) builds the think → act → observe loop.
  3. ask() sends the system prompt plus the user's question, and returns result["messages"][-1].content, the last message, which is the answer.
  4. main() loops on input(). Empty input is skipped, quit/exit/q (or Ctrl+C) ends the program.
Key idea: A tool is a normal Python function with @tool. Its docstring is what the model reads to decide when to use it, so write the docstring clearly.

from langchain_ollama import ChatOllama
from langgraph.prebuilt import create_react_agent
from langchain_core.messages import HumanMessage, SystemMessage
from tools import search_wikipedia

OLLAMA_MODEL = "llama3.2"

SYSTEM_PROMPT = """You are a knowledgeable research assistant with access to Wikipedia.
When a user asks about any topic, person, event, or concept, use the search_wikipedia
tool to get factual information. Summarise clearly, highlight the most important points,
and add context to make the information meaningful. Always mention the Wikipedia source URL."""


def build_agent():
    llm = ChatOllama(model=OLLAMA_MODEL, temperature=0.3)
    return create_react_agent(llm, [search_wikipedia])


def ask(agent, user_input):
    result = agent.invoke({"messages": [SystemMessage(content=SYSTEM_PROMPT), HumanMessage(content=user_input)]})
    return result["messages"][-1].content


def main():
    print("=" * 52)
    print("  📚  Wikipedia Agent  (Ollama + Wikipedia API)")
    print("=" * 52)
    print(f"  Model: {OLLAMA_MODEL}  |  Tool: search_wikipedia")
    print("  Try: 'Who is Marie Curie?' or 'Explain the French Revolution'\n")
    agent = build_agent()
    while True:
        try:
            user_input = input("You: ").strip()
        except (EOFError, KeyboardInterrupt):
            print("\nKeep exploring!"); break
        if not user_input: continue
        if user_input.lower() in ("quit", "exit", "q"):
            print("Keep exploring!"); break
        print("\nAgent:", ask(agent, user_input), "\n")


if __name__ == "__main__":
    main()

2. Air quality agent · tool: get_air_quality(city)

air_quality_agent/tools.py · air_quality_agent/agent.py
What it does

Tells you how clean the air is in a city, with health advice.

How to run
cd day_17_agents\air_quality_agent pip install -r requirements.txt python agent.py
The logic, step by step
  1. Geocoding: the Open-Meteo search API turns the city name into latitude and longitude.
  2. Calls the Open-Meteo Air Quality API for the current European AQI, PM2.5, PM10, NO₂, ozone and CO.
  3. Walks through AQI_LABELS (20 Good, 40 Fair, … 100 Very Poor) and picks the first band the AQI fits into, with its advice.
  4. Returns a formatted text block, and the model explains it in friendly words.

import requests
from langchain_core.tools import tool


AQI_LABELS = [
    (20,  "Good",           "Air quality is excellent. Safe for all outdoor activities."),
    (40,  "Fair",           "Air quality is acceptable. Sensitive groups should monitor."),
    (60,  "Moderate",       "Some pollutants present. Unusually sensitive people may notice."),
    (80,  "Poor",           "Everyone may start to experience health effects."),
    (100, "Very Poor",      "Health warnings. Limit prolonged outdoor exertion."),
    (999, "Extremely Poor", "Health alert — everyone should avoid outdoor activity."),
]


@tool
def get_air_quality(city: str) -> str:
    """
    Get current air quality index (AQI) and pollution levels for any city.
    Returns PM2.5, PM10, ozone, NO₂, CO readings and health advice.
    Uses Open-Meteo Air Quality API — free, no API key.
    """
    # Geocode city
    geo = requests.get(
        "https://geocoding-api.open-meteo.com/v1/search",
        params={"name": city.strip(), "count": 1},
        timeout=10
    ).json()

    if not geo.get("results"):
        return f"City '{city}' not found."

    loc = geo["results"][0]
    lat, lon = loc["latitude"], loc["longitude"]
    full_name = f"{loc['name']}, {loc.get('country', '')}"

    # Fetch air quality
    aq = requests.get(
        "https://air-quality-api.open-meteo.com/v1/air-quality",
        params={
            "latitude": lat,
            "longitude": lon,
            "current": "european_aqi,pm10,pm2_5,carbon_monoxide,nitrogen_dioxide,ozone",
        },
        timeout=10
    ).json()

    cur = aq.get("current", {})
    aqi = cur.get("european_aqi", None)

    label, advice = "Unknown", "No advice available."
    if aqi is not None:
        for threshold, lbl, adv in AQI_LABELS:
            if aqi <= threshold:
                label, advice = lbl, adv
                break

    return (
        f"Air Quality: {full_name}\n"
        f"AQI (European) : {aqi} — {label}\n"
        f"Health Advice  : {advice}\n\n"
        f"PM2.5 : {cur.get('pm2_5', 'N/A')} µg/m³\n"
        f"PM10  : {cur.get('pm10', 'N/A')} µg/m³\n"
        f"NO₂   : {cur.get('nitrogen_dioxide', 'N/A')} µg/m³\n"
        f"Ozone : {cur.get('ozone', 'N/A')} µg/m³\n"
        f"CO    : {cur.get('carbon_monoxide', 'N/A')} µg/m³"
    )

What it does

Answers questions about any country: capital, population, languages, currency and borders.

How to run

The logic, step by step
  1. Calls restcountries.com/v3.1/name/<country>. A non-200 answer means the country wasn't found.
  2. Takes the first result and pulls out each field, using a default of "N/A" whenever one is missing.
  3. Joins the languages and currencies into readable text, and formats the population with commas.

import requests
from langchain_core.tools import tool


@tool
def get_country_info(country: str) -> str:
    """
    Get detailed facts about any country: capital, population, area, languages,
    currency, region, and more. Uses RestCountries API — free, no API key.
    """
    r = requests.get(
        f"https://restcountries.com/v3.1/name/{country.strip()}",
        timeout=10
    )
    if r.status_code != 200:
        return f"Country '{country}' not found. Try the full English name."

    data = r.json()[0]
    capital = (data.get("capital") or ["N/A"])[0]
    population = data.get("population", 0)
    area = data.get("area", 0)
    region = data.get("region", "N/A")
    subregion = data.get("subregion", "N/A")
    languages = ", ".join(data.get("languages", {}).values()) or "N/A"
    currencies = ", ".join(
        f"{v['name']} ({v.get('symbol', '')})"
        for v in data.get("currencies", {}).values()
    ) or "N/A"
    timezones = ", ".join(data.get("timezones", [])[:3])
    un_member = "Yes" if data.get("unMember") else "No"
    borders = ", ".join(data.get("borders", [])) or "None / Island nation"

    return (
        f"Country    : {data['name']['common']} ({data['name']['official']})\n"
        f"Capital    : {capital}\n"
        f"Region     : {region} → {subregion}\n"
        f"Population : {population:,}\n"
        f"Area       : {area:,.0f} km²\n"
        f"Languages  : {languages}\n"
        f"Currency   : {currencies}\n"
        f"Timezones  : {timezones}\n"
        f"Borders    : {borders}\n"
        f"UN Member  : {un_member}"
    )

4. Currency agent · tool: convert_currency(amount, from_currency, to_currency)

currency_agent/tools.py · currency_agent/agent.py
What it does

Converts money between currencies using today's real exchange rate.

How to run
cd day_17_agents\currency_agent pip install -r requirements.txt python agent.py
The logic, step by step
  1. Upper-cases the codes (usd → USD).
  2. Calls api.frankfurter.app/latest?from=USD&to=INR.
  3. If rates is missing, the codes were wrong. Otherwise it computes amount * rate and shows the rate and its date.
  4. The model pulls the three arguments out of a sentence like "How much is 100 dollars in rupees?".

import requests
from langchain_core.tools import tool


@tool
def convert_currency(amount: float, from_currency: str, to_currency: str) -> str:
    """
    Convert an amount between any two currencies using live exchange rates.
    Example: amount=100, from_currency='USD', to_currency='INR'.
    Uses Frankfurter.app — free, no API key, updated daily.
    """
    frm = from_currency.upper().strip()
    to = to_currency.upper().strip()

    r = requests.get(
        "https://api.frankfurter.app/latest",
        params={"from": frm, "to": to},
        timeout=10
    )
    data = r.json()

    if "rates" not in data:
        return f"Could not get exchange rate for {frm} → {to}. Check the currency codes."

    rate = data["rates"][to]
    converted = amount * rate

    return (
        f"{amount:,.2f} {frm} = {converted:,.2f} {to}\n"
        f"Exchange rate : 1 {frm} = {rate} {to}\n"
        f"Rate date     : {data.get('date', 'N/A')}"
    )

import requests
from langchain_core.tools import tool


@tool
def define_word(word: str) -> str:
    """
    Look up the definition, pronunciation, part of speech, usage examples,
    and synonyms for any English word. Uses Free Dictionary API — no API key.
    """
    r = requests.get(
        f"https://api.dictionaryapi.dev/api/v2/entries/en/{word.strip()}",
        timeout=10
    )
    if r.status_code != 200:
        return f"Word '{word}' not found. Check spelling or try a simpler form."

    data = r.json()[0]
    phonetic = data.get("phonetic", "")
    result = f"Word: {data['word']}  {phonetic}\n\n"

    for meaning in data.get("meanings", [])[:3]:
        pos = meaning.get("partOfSpeech", "")
        result += f"[{pos}]\n"
        for defn in meaning.get("definitions", [])[:2]:
            result += f"  Definition : {defn.get('definition', '')}\n"
            if defn.get("example"):
                result += f"  Example    : {defn['example']}\n"
        synonyms = meaning.get("synonyms", [])[:5]
        if synonyms:
            result += f"  Synonyms   : {', '.join(synonyms)}\n"
        result += "\n"

    return result.strip()

import requests
from langchain_core.tools import tool


@tool
def get_joke(category: str = "Any") -> str:
    """
    Get a random joke. Categories: Any, Programming, Misc, Dark, Pun, Spooky, Christmas.
    Uses JokeAPI — free, no API key required.
    """
    valid = ["Any", "Programming", "Misc", "Dark", "Pun", "Spooky", "Christmas"]
    cat = category.strip().title()
    if cat not in valid:
        cat = "Any"

    r = requests.get(
        f"https://v2.jokeapi.dev/joke/{cat}",
        params={"blacklistFlags": "nsfw,explicit,racist,sexist"},
        timeout=10
    )
    data = r.json()

    if data.get("type") == "single":
        return f"😄 {data['joke']}"
    elif data.get("type") == "twopart":
        return f"😄 {data['setup']}\n\n...{data['delivery']} 🥁"
    return "Could not fetch a joke right now. Try again!"


@tool
def get_number_fact(number: str) -> str:
    """
    Get an interesting trivia fact about a number, year, or date.
    Examples: '42', '1969', 'random'. Uses NumbersAPI — free, no API key.
    """
    n = number.strip().lower()
    if n == "random":
        url = "http://numbersapi.com/random/trivia"
    else:
        url = f"http://numbersapi.com/{n}/trivia"

    try:
        r = requests.get(url, timeout=10)
        if r.status_code == 200:
            return f"🔢 {r.text}"
        return f"No fact found for '{number}'."
    except Exception as e:
        return f"Could not fetch fact: {str(e)}"

import requests
from langchain_core.tools import tool


@tool
def search_github_repos(query: str) -> str:
    """
    Search GitHub for public repositories matching a topic, technology, or keyword.
    Returns top 5 results with stars, description, and URL.
    Uses GitHub public API — free, no API key required.
    """
    r = requests.get(
        "https://api.github.com/search/repositories",
        params={"q": query, "sort": "stars", "order": "desc", "per_page": 5},
        headers={"Accept": "application/vnd.github.v3+json"},
        timeout=10
    )
    data = r.json()

    if not data.get("items"):
        return f"No GitHub repositories found for '{query}'."

    total = data.get("total_count", 0)
    result = f"GitHub Search: '{query}' — {total:,} total repos found\nTop 5:\n\n"

    for i, repo in enumerate(data["items"], 1):
        result += (
            f"[{i}] {repo['full_name']}\n"
            f"     ⭐ {repo['stargazers_count']:,} stars  "
            f"🍴 {repo['forks_count']:,} forks  "
            f"[{repo.get('language', 'N/A')}]\n"
            f"     {repo.get('description', 'No description') or 'No description'}\n"
            f"     {repo['html_url']}\n\n"

8. Math tutor agent · tool: solve_math(expression)

math_tutor_agent/tools.py · math_tutor_agent/agent.py
What it does

Solves maths exactly with SymPy (locally, no internet), while the model explains the steps.

How to run
cd day_17_agents\math_tutor_agent pip install -r requirements.txt python agent.py
The logic, step by step
  1. Looks at the first word of the expression to decide what to do.
  2. solve x^2 - 9 → sp.solve(...) (if there is an =, it moves everything to one side first).
  3. diff ... → sp.diff (derivative). integrate ... → sp.integrate (plus C).
  4. Anything else → sympify(...).evalf() for the number and simplify for the tidy form.
  5. Any error returns a hint ("try 'solve x^2 - 4'") instead of crashing.
Key idea: The model is bad at arithmetic but good at explaining. The tool does the exact maths.
Show the full source of math_tutor_agent/tools.py (55 lines)
from langchain_core.tools import tool


@tool
def solve_math(expression: str) -> str:
    """
    Solve or evaluate any mathematical expression, equation, derivative, or integral.
    Examples:
      '2 + 2'            → arithmetic
      'sqrt(144)'        → functions
      'solve x^2 - 9'    → find roots
      'diff x^3 + 2*x'   → derivative
      'integrate sin(x)' → integral
    Uses SymPy — runs locally, no internet needed.
    """
    import sympy as sp

    x, y, z = sp.symbols("x y z")
    expr = expression.strip()
    lower = expr.lower()

    try:
        if lower.startswith("solve"):
            eq_str = lower.replace("solve", "", 1).strip()
            if "=" in eq_str:
                lhs, rhs = eq_str.split("=", 1)
                eq = sp.sympify(lhs) - sp.sympify(rhs)
            else:
                eq = sp.sympify(eq_str)
            solutions = sp.solve(eq, x)
            return f"Equation : {expr}\nSolutions: x = {solutions}"

        elif lower.startswith(("diff", "derivative")):
            expr_str = lower.replace("derivative", "").replace("diff", "", 1).strip()
            result = sp.diff(sp.sympify(expr_str), x)
            simplified = sp.simplify(result)
            return f"d/dx [ {expr_str} ] = {simplified}"

        elif lower.startswith(("integrate", "integral")):
            expr_str = lower.replace("integral", "").replace("integrate", "", 1).strip()
            result = sp.integrate(sp.sympify(expr_str), x)
            return f"∫ {expr_str} dx = {result} + C"

        else:
            parsed = sp.sympify(expr)
            evaluated = parsed.evalf()
            simplified = sp.simplify(parsed)
            return (
                f"Expression : {expr}\n"
                f"Result     : {evaluated}\n"
                f"Simplified : {simplified}"
            )

    except Exception as e:
        return f"Could not evaluate '{expr}': {str(e)}\nTry rewording, e.g. 'solve x^2 - 4'"

import yfinance as yf
from langchain_core.tools import tool


@tool
def get_stock_price(ticker: str) -> str:
    """
    Get the current stock price, today's change, volume, and 52-week range
    for any stock ticker symbol (e.g. AAPL, TSLA, INFY.NS, RELIANCE.NS).
    Uses yfinance — free, no API key required.
    """
    symbol = ticker.upper().strip()
    try:
        stock = yf.Ticker(symbol)
        hist = stock.history(period="2d")
        info = stock.info

        if hist.empty:
            return f"No data found for ticker '{symbol}'. Check the symbol."

        current = hist["Close"].iloc[-1]
        prev_close = hist["Close"].iloc[-2] if len(hist) >= 2 else hist["Open"].iloc[-1]
        change = current - prev_close
        pct = (change / prev_close) * 100
        volume = hist["Volume"].iloc[-1]

        return (
            f"Stock : {info.get('longName', symbol)} ({symbol})\n"
            f"Price : ${current:.2f}\n"
            f"Change: {change:+.2f} ({pct:+.2f}%) vs previous close\n"
            f"Volume: {volume:,.0f}\n"
            f"52w High : ${info.get('fiftyTwoWeekHigh', 'N/A')}\n"
            f"52w Low  : ${info.get('fiftyTwoWeekLow', 'N/A')}\n"
            f"Market Cap: ${info.get('marketCap', 0):,.0f}\n"
            f"Sector: {info.get('sector', 'N/A')}"
        )
    except Exception as e:
        return f"Could not fetch data for '{symbol}': {str(e)}"

import requests
from langchain_core.tools import tool


WMO_CODES = {
    0: "Clear sky", 1: "Mainly clear", 2: "Partly cloudy", 3: "Overcast",
    45: "Foggy", 48: "Icy fog", 51: "Light drizzle", 53: "Moderate drizzle",
    61: "Light rain", 63: "Moderate rain", 65: "Heavy rain",
    71: "Light snow", 73: "Moderate snow", 75: "Heavy snow",
    80: "Rain showers", 81: "Moderate showers", 82: "Heavy showers",
    95: "Thunderstorm", 96: "Thunderstorm with hail",
}


@tool
def get_destination_info(city: str) -> str:
    """
    Get current weather and 3-day travel forecast for any destination city.
    Returns temperature, conditions, humidity, wind speed, and travel tips.
    Uses free Open-Meteo API — no API key required.
    """
    # Step 1: Geocode city → lat/lon
    geo_resp = requests.get(
        "https://geocoding-api.open-meteo.com/v1/search",
        params={"name": city, "count": 1, "language": "en", "format": "json"},
        timeout=10,
    )
    geo_data = geo_resp.json()

    if not geo_data.get("results"):
        return f"Sorry, I could not find the location '{city}'. Please check the city name."

    loc = geo_data["results"][0]
    lat, lon = loc["latitude"], loc["longitude"]
    country = loc.get("country", "Unknown")
    full_name = f"{loc['name']}, {country}"

    # Step 2: Fetch current weather + 3-day forecast
    weather_resp = requests.get(
        "https://api.open-meteo.com/v1/forecast",
        params={
            "latitude": lat,
            "longitude": lon,
            "current": "temperature_2m,relative_humidity_2m,wind_speed_10m,weathercode,apparent_temperature",
            "daily": "temperature_2m_max,temperature_2m_min,precipitation_probability_max,weathercode",
            "timezone": "auto",
            "forecast_days": 3,
        },
        timeout=10,
    )
    data = weather_resp.json()

    cur = data.get("current", {})
    daily = data.get("daily", {})

    temp = cur.get("temperature_2m", "N/A")
    feels = cur.get("apparent_temperature", "N/A")
    humidity = cur.get("relative_humidity_2m", "N/A")
    wind = cur.get("wind_speed_10m", "N/A")
    condition = WMO_CODES.get(cur.get("weathercode", 0), "Unknown")

    # Build 3-day forecast string
    forecast_lines = []
    for i in range(min(3, len(daily.get("time", [])))):
        date = daily["time"][i]
        hi = daily["temperature_2m_max"][i]
        lo = daily["temperature_2m_min"][i]
        rain = daily["precipitation_probability_max"][i]
        cond = WMO_CODES.get(daily["weathercode"][i], "Unknown")
        forecast_lines.append(f"  {date}: {lo}°C – {hi}°C | {cond} | Rain: {rain}%")

    # Travel tip based on current temp
    try:
        t = float(temp)
        if t < 5:
            tip = "Very cold — pack heavy winter clothing."
        elif t < 15:
            tip = "Cool weather — bring a jacket."
        elif t <= 28:
            tip = "Pleasant weather — great for sightseeing!"
        elif t <= 35:
            tip = "Warm — stay hydrated and use sunscreen."
        else:
            tip = "Extremely hot — limit outdoor activities midday."
    except (ValueError, TypeError):
        tip = "Check local conditions before travelling."

    forecast_text = "\n".join(forecast_lines) if forecast_lines else "  Not available"

    return (
        f"=== Travel Info: {full_name} ===\n"
        f"Current Conditions: {condition}\n"
        f"Temperature: {temp}°C (feels like {feels}°C)\n"
        f"Humidity: {humidity}%  |  Wind: {wind} km/h\n\n"
        f"3-Day Forecast:\n{forecast_text}\n\n"
        f"Travel Tip: {tip}"
    )

def search_wikipedia(query: str) -> str:

            page = wikipedia.page(results[0], auto_suggest=False)
            summary = wikipedia.summary(results[0], sentences=5, auto_suggest=False)
        except wikipedia.exceptions.DisambiguationError as e:
            page = wikipedia.page(e.options[0], auto_suggest=False)
            summary = wikipedia.summary(e.options[0], sentences=5, auto_suggest=False)

        return (
            f"Wikipedia: {page.title}\n"
            f"URL      : {page.url}\n\n"
            f"{summary}"
        )
    except Exception as e:
        return f"Could not fetch Wikipedia page: {str(e)}"

weather_agent/main.py
What it does

Predicts tomorrow's maximum temperature from the last 7 real days, shows every think/act/observe step, and then compares the guess with the official forecast.

How to run
cd day_17_agents\weather_agent python main.py Hyderabad
The logic, step by step
  1. The helpers find_city and fetch_daily (plain Python, not tools) do the geocoding and fetch the daily max/min temperatures from Open-Meteo.
  2. Tool 1, get_past_temperatures(city): the last 7 days plus today, one line per day.
  3. Tool 2, temperature_trend(city): fits a straight line through the max temperatures (least-squares slope) and returns °C per day, the next value on the line, and the average of the last 3 days.
  4. It fetches the numbers itself, because small models often break when copying lists of numbers into a tool call.
  5. The system prompt fixes the order: call tool 1, then tool 2, then answer with one number. Never invent temperatures.
  6. agent.stream(..., stream_mode="values") gives the growing message list after each step. The loop prints only the new messages: AI with tool calls (THINK → ACT), tool messages (OBSERVE), and the final AI answer.
  7. Last, official_forecast() prints Open-Meteo's own forecast for tomorrow, so you can check the agent.

"""Weather Prediction Agent · LangChain + Ollama

The agent gets the last 7 days of real temperatures for a city from a free
weather API (Open-Meteo, no API key), works out the trend, and predicts
tomorrow's max temperature. At the end we compare its guess with
Open-Meteo's official forecast.

Run (with the venv active):
    python main.py                 # asks for a city
    python main.py Hyderabad       # or pass the city directly

Needs Ollama running with:  ollama pull llama3.2
"""

import sys
import requests
from langchain.agents import create_agent
from langchain_core.tools import tool
from langchain_ollama import ChatOllama

GEO_URL = "https://geocoding-api.open-meteo.com/v1/search"
WEATHER_URL = "https://api.open-meteo.com/v1/forecast"
MODEL = "llama3.2"


# ---------- helpers (plain Python, not tools) ----------

def find_city(city: str) -> dict:
    """Turn a city name into latitude/longitude."""
    r = requests.get(GEO_URL, params={"name": city, "count": 1}, timeout=15)
    r.raise_for_status()
    results = r.json().get("results")
    if not results:
        raise ValueError(f"City '{city}' not found")
    return results[0]


def fetch_daily(city: str, past_days: int, forecast_days: int) -> dict:
    """Get daily max/min temperatures from Open-Meteo."""
    place = find_city(city)
    r = requests.get(WEATHER_URL, params={
        "latitude": place["latitude"],
        "longitude": place["longitude"],
        "daily": "temperature_2m_max,temperature_2m_min",
        "past_days": past_days,
        "forecast_days": forecast_days,
        "timezone": "auto",
    }, timeout=15)
    r.raise_for_status()
    return {"place": place, "daily": r.json()["daily"]}


# ---------- TOOLS: the agent's "hands" ----------

@tool
def get_past_temperatures(city: str) -> str:
    """Get the real daily max and min temperatures (°C) for the last 7 days
    in a city, plus today. Use this first to see recent weather."""
    data = fetch_daily(city, past_days=7, forecast_days=1)
    d, p = data["daily"], data["place"]
    lines = [f"{p['name']}, {p.get('country', '')}:"]
    for day, tmax, tmin in zip(d["time"], d["temperature_2m_max"], d["temperature_2m_min"]):
        lines.append(f"{day}: max {tmax}°C, min {tmin}°C")
    return "\n".join(lines)


@tool
def temperature_trend(city: str) -> str:
    """Calculate the trend of the last 7 days of max temperatures in a city:
    how much it changes per day and the next value on the trend line.
    Use this to do the maths instead of guessing."""
    # The tool fetches the numbers itself. Small local models (llama3.2 3B)
    # often break when they have to copy a list of numbers into a tool call.
    max_temps = fetch_daily(city, past_days=7, forecast_days=1)["daily"]["temperature_2m_max"]
    max_temps = [t for t in max_temps if t is not None]
    n = len(max_temps)
    if n < 2:
        return "Need at least 2 temperatures."
    xs = range(n)
    x_mean = sum(xs) / n
    y_mean = sum(max_temps) / n
    slope = sum((x - x_mean) * (y - y_mean) for x, y in zip(xs, max_temps)) / \
            sum((x - x_mean) ** 2 for x in xs)
    next_value = y_mean + slope * (n - x_mean)
    avg_last3 = sum(max_temps[-3:]) / min(3, n)
    return (f"trend: {slope:+.2f} °C per day | "
            f"straight-line next value: {next_value:.1f} °C | "
            f"average of last 3 days: {avg_last3:.1f} °C")


# ---------- THE AGENT: brain (LLM) + hands (tools) + rules (system prompt) ----------

agent = create_agent(
    model=ChatOllama(model=MODEL, temperature=0),
    tools=[get_past_temperatures, temperature_trend],
    system_prompt=(
        "You are a weather analyst. To predict tomorrow's max temperature:\n"
        "1. Call get_past_temperatures for the city.\n"
        "2. Call temperature_trend for the same city.\n"
        "3. Answer with: the predicted max temperature for tomorrow (one number in °C), "
        "and one or two short sentences explaining why, based on the data.\n"
        "Never invent temperatures. Only use numbers that came from the tools."
    ),
)


def official_forecast(city: str) -> str:
    """Open-Meteo's own forecast for tomorrow, to check the agent's guess."""
    d = fetch_daily(city, past_days=0, forecast_days=2)["daily"]
    return f"{d['time'][1]}: max {d['temperature_2m_max'][1]}°C, min {d['temperature_2m_min'][1]}°C"


if __name__ == "__main__":
    city = " ".join(sys.argv[1:]) or input("City: ").strip() or "Hyderabad"
    question = f"Predict tomorrow's maximum temperature in {city}."
    print(f"\n🧑 You: {question}\n")

    # stream_mode="values" shows each step of the loop as it happens
    seen = 1  # skip our own question (message 0)
    for step in agent.stream({"messages": [{"role": "user", "content": question}]},
                             stream_mode="values"):
        # the agent may call several tools at once, so print every new message
        for msg in step["messages"][seen:]:
            if msg.type == "ai" and msg.tool_calls:
                for call in msg.tool_calls:
                    print(f"🧠 THINK → 🛠️ ACT: call {call['name']}({call['args']})")
            elif msg.type == "tool":
                print(f"🔎 OBSERVE ({msg.name}):\n{msg.content}\n")
            elif msg.type == "ai":
                print(f"🤖 Agent's prediction:\n{msg.content}\n")
        seen = len(step["messages"])

    print(f"📡 Official Open-Meteo forecast for tomorrow → {official_forecast(city)}")
